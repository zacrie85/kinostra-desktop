    89→   110→  const base = ['avc1.640028', 'avc1.4D0028', 'avc1.42002A'];
    90→   111→  const mkHw = c => ({ codec: c, width: W, height: H, bitrate: br, framerate: fps, hardwareAcceleration: 'prefer-hardware' });

    94→   115→

   124→   145→    const hw = (state.acc === 'auto' && _hwDemoted) ? null : await probeKind(mkHw, 'hw');

   136→   157→      } catch (e) { }
   137→   158→    }
   138→   159→  }
   139→   160→  if (res) {
   140→   161→    _accCache = { key, res };
   141→   162→    _lastEnc = res.enc;
   142→   163→    return res.cfg;
   143→   164→  }
   144→   165→  return null;
   145→   166→}

   197→   218→async function renderMix(start, durOut, spd = 1) {
   198→   219→  const sr = 48000, len = Math.max(1, Math.ceil(durOut * sr));
   199→   220→  const oc = new OfflineAudioContext(2, len, sr);
   200→   221→  /* v3.0: AUDIO SEHAT — rantai perbaikan: filter desis (opsional) →
   201→   222→     normalizer (RMS diukur sekali per video) → kompresor lembut.
   202→   223→     Bila mati, sumber tersambung langsung ke destination (dulu). */
   203→   224→  let sink = null;
   204→   225→  if (state.audioFix && state.audioFix.on) {
   205→   226→    sink = oc.createGain();
   206→   227→    let cur = sink;
   207→   228→    if (state.audioFix.hiss) {
   208→   229→      const hp = oc.createBiquadFilter(); hp.type = 'highpass'; hp.frequency.value = 95; hp.Q.value = 0.6;
   209→   230→      const lp = oc.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 11500;
   210→   231→      cur.connect(hp); hp.connect(lp); cur = lp;
   211→   232→    }
   212→   233→    const norm = oc.createGain();
   213→   234→    const rms = (typeof measureAudioRms === 'function') ? measureAudioRms() : 0;
   214→   235→    norm.gain.value = rms > 0.004 ? clamp(0.16 / rms, 0.5, 4) : 1;
   215→   236→    cur.connect(norm);
   216→   237→    const comp = oc.createDynamicsCompressor();
   217→   238→    comp.threshold.value = -18; comp.knee.value = 20; comp.ratio.value = 2.6;
   218→   239→    comp.attack.value = 0.01; comp.release.value = 0.24;
   219→   240→    norm.connect(comp); comp.connect(oc.destination);
   220→   241→  }
   221→   242→  const connectSrc = node => { if (sink) node.connect(sink); else node.connect(oc.destination); };
   222→   243→  /* v2.10 FIX: dulu s.start(0, start, durBuffer − start) melempar
   223→   244→     InvalidStateError ketika start > durasi buffer (mis. WAKTU MULAI 1:05
   224→   245→     padahal skor musik masih milik video sebelumnya yang lebih pendek) —
   225→   246→     inilah pemicu utama batch ekspor "selalu dibatalkan" sejak v2.9.
   226→   247→     Sekarang: audio asli dilewati bila sudah lewat, musik diputar BERULANG
   227→   248→     (loop) sehingga durasi buffer berapa pun tidak pernah crash.
   228→   249→     v2.12: kecepatan → audio asli diputar lebih cepat/lambat memakai
   229→   250→     playbackRate (pitch ikut naik/turun seperti efek fast-forward klasik). */
   230→   251→  if (state.audioBuffer && state.audioBuffer.duration > 0.01 && start < state.audioBuffer.duration - 0.01) {
   231→   252→    const s = oc.createBufferSource(); s.buffer = state.audioBuffer;
   232→   253→    s.playbackRate.value = spd;
   233→   254→    const g = oc.createGain(); g.gain.value = state.audioGain;
   234→   255→    s.connect(g); connectSrc(g);
   235→   256→    s.start(0, start, Math.max(0.01, Math.min(durOut, (state.audioBuffer.duration - start) / spd)));
   236→   257→  }
   237→   258→  if (state.music.buffer && state.music.buffer.duration > 0.01) {
   238→   259→    const s = oc.createBufferSource(); s.buffer = state.music.buffer;
   239→   260→    s.playbackRate.value = spd;
   240→   261→    const g = oc.createGain(); g.gain.value = state.music.gain;
   241→   262→    s.connect(g); connectSrc(g);
   242→   263→    s.loop = true;   /* buffer lebih pendek dari part → mengulang, bukan crash */
   243→   264→    s.start(0, start % state.music.buffer.duration, durOut);
   244→   265→  }
   245→   266→  /* v3.0: TTS NARASI — suara AI offline diletakkan di waktu media
   246→   267→     state.tts.at; ikut kecepatan video (spd). Potongan di luar part
   247→   268→     dilewati; bagian yang terpotong di awal part disesuaikan. */

   250→   271→    let outOffN = (T.at - start) / spd, srcOff = 0;
   251→   272→    if (outOffN < 0) { srcOff = -outOffN * spd; outOffN = 0; }
   252→   273→    if (outOffN < durOut - 0.01 && srcOff < T.buffer.duration) {
   253→   274→      const s = oc.createBufferSource(); s.buffer = T.buffer;
   254→   275→      s.playbackRate.value = spd;
   255→   276→      const g = oc.createGain(); g.gain.value = clamp(T.gain == null ? 0.9 : T.gain, 0, 1.6);
   256→   277→      s.connect(g); connectSrc(g);
   257→   278→      s.start(outOffN, srcOff, Math.max(0.01, Math.min(durOut - outOffN, (T.buffer.duration - srcOff) / spd)));
   258→   279→    }
   259→   280→  }

   271→   292→async function encodeAudioStream(muxer, seg, durOut, spd = 1, codec, mergeOff = 0) {
   272→   293→  const sr = 48000, chs = 2;
   273→   294→  const aenc = new AudioEncoder({ output: (c, m) => muxer.addAudioChunk(c, m), error: e => console.warn('aenc', e) });
   274→   295→  aenc.configure({ codec, sampleRate: sr, numberOfChannels: chs, bitrate: 160000 });
   275→   296→  try {
   276→   297→    const BLK = 60;
   277→   298→    const fbuf = new Float32Array(sr * chs);
   278→   299→    let outOff = 0;
   279→   300→    while (outOff < durOut - 1e-6) {
   280→   301→      if (state.abort) throw new Error('Dibatalkan');
   281→   302→      if (state.exportFatal) throw state.exportFatal;
   282→   303→      const len = Math.min(BLK, durOut - outOff);
   283→   304→      const mix = await renderMix(seg.start + outOff * spd, len, spd);
   284→   305→      const L = mix.getChannelData(0), R2 = chs > 1 ? mix.getChannelData(1) : L;
   285→   306→      for (let off = 0; off < mix.length; off += sr) {
   286→   307→        if (state.abort) throw new Error('Dibatalkan');
   287→   308→        if (state.exportFatal) throw state.exportFatal;
   288→   309→        const n = Math.min(sr, mix.length - off);
   289→   310→        fbuf.set(L.subarray(off, off + n), 0);
   290→   311→        if (chs > 1) fbuf.set(R2.subarray(off, off + n), n);
   291→   312→        const ad = new AudioData({ format: 'f32-planar', sampleRate: sr, numberOfFrames: n, numberOfChannels: chs,
   292→   313→          timestamp: Math.round((mergeOff + outOff + off / sr) * 1e6), data: fbuf });
   293→   314→        aenc.encode(ad); ad.close();
   294→   315→        let drainMs = 0;
   295→   316→        while (aenc.encodeQueueSize > 10) { await sleep(2); if (state.abort) throw new Error('Dibatalkan'); if (state.exportFatal) throw state.exportFatal; if ((drainMs += 2) > 30000) throw new Error('encoder audio macet'); }
   296→   317→      }
   297→   318→      outOff += len;
   298→   319→    }
   299→   320→    await aenc.flush();
   300→   321→  } finally {
   301→   322→    try { aenc.close(); } catch (e) { }   /* v2.15: sesi encoder audio selalu ditutup */
   302→   323→  }

   328→   349→    : Math.max(1, Math.min(4, parts, Math.floor(cores / 3)));
   329→   350→}

   331→   352→/* ---------- FALLBACK: rantai seek per frame (lama) ----------
   332→   353→   Dipakai untuk audio-only (tanpa frame video) dan bila browser
   333→   354→   tanpa requestVideoFrameCallback. noSeek = audio (tidak perlu seek)
   334→   355→   v2.12: spd = kecepatan — frame ke-i = waktu output i/fps → sumber
   335→   356→   i*spd/fps; PTS output tetap i*1e6/fps (monoton rapat). */
   336→   357→async function renderFramesBySeek(v, ecv, ec, venc, seg, fps, slot, doneCb, noSeek = false, spd = 1, tsOff = 0) {
   337→   358→  const n = Math.max(1, Math.round((seg.end - seg.start) / spd * fps));
   338→   359→  let seekP = null;
   339→   360→  for (let i = 0; i < n; i++) {
   340→   361→    if (state.abort) throw new Error('Dibatalkan');
   341→   362→    if (state.exportFatal) throw state.exportFatal;
   342→   363→    if (seekP) { await seekP; seekP = null; }
   343→   364→    const t = Math.min(seg.start + i * spd / fps, Math.max(0, (v.duration || seg.end) - 0.011));
   344→   365→    setCompSrc(v);
   345→   366→    drawComposition(ec, ecv.width, ecv.height, t);
   346→   367→    setCompSrc(null);
   347→   368→    const vf = new VideoFrame(ecv, { timestamp: Math.round(tsOff + i * 1e6 / fps), duration: Math.round(1e6 / fps) });
   348→   369→    venc.encode(vf, { keyFrame: i % (fps * 2) === 0 }); vf.close();
   349→   370→    if (!noSeek && i + 1 < n) seekP = seekTo(seg.start + (i + 1) * spd / fps, v);
   350→   371→    while (venc.encodeQueueSize > 14) await sleep(2);
   351→   372→    slot.fr = i + 1;
   352→   373→    doneCb && doneCb();
   353→   374→  }
   354→   375→  return n;
   355→   376→}

   357→   378→/* ---------- WORKER SATU PART (v2.2 TURBO) ----------

   360→   381→async function renderPartWorker(seg, srcUrl, W, H, fps, br, vCfg, aCodec, slot, doneCb, mergeCtx = null) {

   429→   450→        video: mkVideo, ...mkAudio
   430→   451→      });

   433→   454→      _stage = 'audio';
   434→   455→      /* v2.15: encode audio PER BLOK — RAM datar di part panjang
   435→   456→         v3.0 OPTIMASI KECEPATAN: audio di-encode BERSAMAAN dengan render
   436→   457→         video (dulu audio selesai dulu → baru video: waktu menjumlah).
   437→   458→         Kedua encoder mengalir ke muxer yang sama — mp4-muxer menerima
   438→   459→         potongan video/audio ter-selang; hasil identik, ±30-45% lebih cepat. */

   440→   461→        encodeAudioStream(muxer, seg, durS, spd, aCodec, mergeCtx ? mergeCtx.off : 0),
   441→   462→        300000, 'encoding audio macet'
   442→   463→      ).catch(e => { audioErr = audioErr || e; });
   443→   464→    }
   444→   465→    _stage = 'configure';
   445→   466→    const hw = vCfg.hardwareAcceleration === 'prefer-hardware';
   446→   467→    /* v2.7 FIX "SUARA JALAN, GAMBAR BEKU": encoder GPU Windows (Media
   447→   468→       Foundation) dengan latencyMode 'quality' dapat mengeluarkan chunk
   448→   469→       B-frame berurutan DECODE (timestamp bolak-balik). mp4-muxer melempar
   449→   470→       error monotonic DI DALAM callback output → pengecualian tertelan
   450→   471→       diam-diam → hanya beberapa frame pertama masuk MP4 → gambar beku +
   451→   472→       audio normal + ekspor "berhasil". Perisai 4 lapis:
   452→   473→       1) latencyMode 'realtime' utk SEMUA encoder (tanpa reorder B-frame)
   453→   474→       2) timestamp turun > 5 ms terdeteksi → gagal cepat → auto-retry SW
   454→   475→       3) error encoder diteruskan ke promise (bukan hanya console)
   455→   476→       4) verifikasi jumlah chunk video setelah flush (≥ 90% frame) */
   456→   477→    let vChunks = 0, lastVts = -Infinity, vencErrMsg = null;
   457→   478→    venc = new VideoEncoder({
   458→   479→      output: (c, m) => {
   459→   480→        const ts = typeof c.timestamp === 'number' ? c.timestamp : 0;
   460→   481→        if (vChunks > 0 && ts <= lastVts) {
   461→   482→          if (lastVts - ts > 5000) {
   462→   483→            vencErrMsg = `encoder mengeluarkan frame dengan urutan waktu tidak naik (${Math.round(lastVts / 1000)} → ${Math.round(ts / 1000)} ms)`;
   463→   484→            console.error('venc-stream', vencErrMsg);
   464→   485→          }
   465→   486→          return; /* jangan masukkan chunk rusak/nol-durasi ke muxer */
   466→   487→        }
   467→   488→        lastVts = ts; vChunks++;
   468→   489→        muxer.addVideoChunk(c, m);
   469→   490→      },
   470→   491→      error: e => { vencErrMsg = (e && e.message) || String(e); console.error('venc', vencErrMsg); }
   471→   492→    });
   472→   493→    venc.configure({ ...vCfg, latencyMode: 'realtime' });
   473→   494→    const CAP = hw ? 40 : 14;
   474→   495→
   475→   496→    /* gambar 1 frame pada waktu media m → encode dengan timestamp nyata
   476→   497→       v2.12: PTS = waktu OUTPUT = (m − mulai part) ÷ kecepatan
   477→   498→       v3.0: + offset global mergeCtx.off saat GABUNG VIDEO */
   478→   499→    const drawAt = (t, ts) => {
   479→   500→      setCompSrc(v);
   480→   501→      drawComposition(ec, W, H, t);
   481→   502→      setCompSrc(null);
   482→   503→      const vf = new VideoFrame(ecv, { timestamp: ts, duration: Math.round(1e6 / fps) });
   483→   504→      venc.encode(vf, { keyFrame: n % (fps * 2) === 0 }); vf.close();
   484→   505→      n++; slot.fr = n;
   485→   506→    };
   486→   507→    const outTs = (m) => Math.max(0, Math.round((((mergeCtx ? mergeCtx.off : 0) + (m - seg.start) / spd) * 1e6)));
   487→   508→
   488→   509→    if (RVFC_OK && !state.isAudio) {
   489→   510→      /* ============ v2.2: CAPTURE PLAYBACK (tanpa seek per frame) ============
   490→   511→         v2.5: + WATCHDOG ANTI-MACET. Penyebab lama ekspor "macet di bagian
   491→   512→         akhir tanpa pesan": frame callback (rVFC) berhenti memicu di frame
   492→   513→         terakhir & event ended tidak sampai, atau encoder GPU berhenti
   493→   514→         mengeluarkan output → promise render tidak pernah selesai.
   494→   515→         Sekarang interval 250 ms memantau: video selesai? waktu media
   495→   516→         sudah mencapai akhir part? video error? encoder diam terlalu
   496→   517→         lama? → semua diarahkan SELESAI atau GAGAL DENGAN PESAN JELAS. */
   497→   518→      _stage = 'stream';
   498→   519→      /* v2.11 FIX "KADANG GERAK KADANG BEKU": dulu capRate mulai 2× dan bisa
   499→   520→         naik sampai 4× — di atas kemampuan decoder/compositor, frame sumber
   500→   521→         dibuang beramai-ramai, mediaTime antar frame tersaji LOMPAT, PTS
   501→   522→         ekspor ikut bolong → pemutar menahan frame lama berdetik-detik.
   502→   523→         Sekarang: mulai 1.5×, turun SEGERA saat ada frame hilang, dan setiap
   503→   524→         lompatan DIISI ULANG dengan seek presisi (refillGap) sehingga aliran
   504→   525→         PTS selalu rapat & mulus.
   505→   526→         v2.14: bila encoder GPU (hardware) aktif, langit-langit dinaikkan
   506→   527→         ke 3× — encoding bukan lagi hambatan, decoder-lah penentu. */
   507→   528→      const RATE_MAX = hw ? 3 : (_xPar >= 3 ? 2 : 2.5);
   508→   529→      /* v2.12: toleransi & target lompatan diskalakan kecepatan —
   509→   530→         pada spd 2× jarak sumber antar frame tampil 2× lebih renggang */
   510→   531→      const frameGapTol = 1.8 / fps * spd;
   511→   532→      let capRate = 1.5, dAcc = 0, dN = 0, lastM = -1, hadFirst = false;
   512→   533→      const applyRate = () => { try { v.playbackRate = Math.min(4.2, capRate * spd); } catch (e) { } };
   513→   534→      applyRate();
   514→   535→      let lastAct = Date.now(), qStallSince = 0;
   515→   536→      let refilling = false, refillN = 0, maxGapMs = 0;
   516→   537→      /* v2.11: isi ulang frame yang hilang di interval (from, to) — seek
   517→   538→         presisi slot per slot, encode dengan PTS aslinya. Dipanggil SEBELUM
   518→   539→         frame@to di-encode agar urutan PTS tetap naik. lastAct disegarkan
   519→   540→         tiap slot supaya watchdog 30 dtk tidak mengira render diam. */
   520→   541→      const refillGap = async (from, to) => {
   521→   542→        const step = spd / fps;
   522→   543→        let t = from + step, k = 0;
   523→   544→        while (t < to - step * 0.5) {
   524→   545→          if (state.abort) throw new Error('Dibatalkan');
   525→   546→          if (state.exportFatal) throw state.exportFatal;
   526→   547→          await seekTo(Math.min(t, Math.max(0, (v.duration || seg.end) - 0.011)), v);
   527→   548→          drawAt(t, outTs(t));
   528→   549→          lastAct = Date.now(); k++; t += step;
   529→   550→          doneCb && doneCb();
   530→   551→        }
   531→   552→        refillN += k; return k;
   532→   553→      };
   533→   554→      await new Promise((resolve, rej) => {
   534→   555→        let finished = false;
   535→   556→        const stop = () => { try { v.pause(); } catch (e) { } };
   536→   557→        const fin = () => { if (finished) return; finished = true; stop(); clearInterval(watch); resolve(); };
   537→   558→        const fail = (err) => { if (finished) return; finished = true; stop(); clearInterval(watch); rej(err); };
   538→   559→        /* --- v2.5: WATCHDOG (detak 250 ms) --- */
   539→   560→        const watch = setInterval(() => {
   540→   561→          if (finished) { clearInterval(watch); return; }
   541→   562→          if (state.abort) return fail(new Error('Dibatalkan'));
   542→   563→          if (state.exportFatal) return fail(state.exportFatal);   /* v2.10: part lain sudah gagal */
   543→   564→          /* v2.7: error encoder terdeteksi → gagal cepat → retry software */
   544→   565→          if (vencErrMsg) return fail(new Error('encoder video bermasalah: ' + vencErrMsg));
   545→   566→          if (v.error) return fail(new Error('video rusak/tak terbaca: ' + (v.error.message || ('kode ' + v.error.code))));
   546→   567→          /* KUNCI FIX: tangani akhir video walau frame callback mati —
   547→   568→             inilah yang dulu bikin macet selamanya di bagian akhir */
   548→   569→          if (v.ended || v.currentTime >= seg.end - 0.004) return fin();
   549→   570→          const now = Date.now();
   550→   571→          if (venc.encodeQueueSize > CAP) {
   551→   572→            if (!qStallSince) qStallSince = now;
   552→   573→            else if (now - qStallSince > 45000)
   553→   574→              return fail(new Error('encoder video macet (antrean tidak mengalir 45 dtk) — akan dicoba ulang otomatis dengan encoder software'));
   554→   575→          } else qStallSince = 0;
   555→   576→          if (now - lastAct > 30000)
   556→   577→            return fail(new Error('render diam total 30 dtk — akan dicoba ulang otomatis dengan encoder software'));
   557→   578→          /* dorong ulang bila video diam padahal encoder tidak penuh */
   558→   579→          if (v.paused && !refilling && !v.seeking && venc.encodeQueueSize <= CAP && !v.ended) {
   559→   580→            try { const p = v.play(); if (p && p.catch) p.catch(err => fail(new Error('video tidak bisa dilanjut: ' + (err.message || err)))); }
   560→   581→            catch (err) { fail(new Error('video tidak bisa dilanjut: ' + (err.message || err))); }
   561→   582→          }
   562→   583→        }, 250);
   563→   584→        const onFrame = async (now, meta) => {
   564→   585→          try {
   565→   586→            if (finished) return;
   566→   587→            lastAct = Date.now();
   567→   588→            if (state.abort) return fail(new Error('Dibatalkan'));
   568→   589→            if (state.exportFatal) return fail(state.exportFatal);   /* v2.10 */
   569→   590→            const m = meta.mediaTime;
   570→   591→            if (v.ended || m >= seg.end - 0.004) return fin();
   571→   592→            if (m >= seg.start - 0.06 && m > lastM) {
   572→   593→              /* v2.11: DETEKSI FRAME HILANG — lompatan mediaTime > 1.8× durasi
   573→   594→                 frame berarti decoder/compositor membuang frame. Isi slot yang
   574→   595→                 bolong lewat seek presisi SEBELUM frame m di-encode supaya
   575→   596→                 urutan PTS rapat (tanpa durasi sampel multi-detik = beku). */
   576→   597→              const gap = hadFirst ? m - lastM : 0;
   577→   598→              if (gap > frameGapTol) {
   578→   599→                maxGapMs = Math.max(maxGapMs, gap * 1000);
   579→   600→                refilling = true; stop();
   580→   601→                const k = await refillGap(lastM, m);
   581→   602→                if (finished) return;
   582→   603→                await seekTo(Math.min(m, Math.max(0, (v.duration || seg.end) - 0.011)), v);
   583→   604→                refilling = false;
   584→   605→                if (finished) return;
   585→   606→                /* decoder kewalahan pada rate ini → turun SEGERA */
   586→   607→                if (capRate > 1) { capRate = Math.max(1, capRate - 0.5); applyRate(); }
   587→   608→                dAcc = 0; dN = 0;
   588→   609→              }
   589→   610→              drawAt(m, outTs(m));
   590→   611→              hadFirst = true;
   591→   612→              /* rate adaptif: naik HANYA bila jendela bersih tanpa lompatan
   592→   613→                 v2.12: target jarak antar frame = spd/fps (waktu sumber) */
   593→   614→              if (lastM >= 0) {
   594→   615→                const d = m - lastM;
   595→   616→                if (d > 0 && d < 0.6 * spd) {
   596→   617→                  dAcc += d; dN++;
   597→   618→                  if (dN >= 24) {
   598→   619→                    const avg = dAcc / dN; dAcc = 0; dN = 0; const tg = spd / fps;
   599→   620→                    if (avg > tg * 1.5 && capRate > 1) { capRate = Math.max(1, capRate - 0.5); applyRate(); }
   600→   621→                    else if (avg < tg * 1.18 && capRate < RATE_MAX) { capRate = Math.min(RATE_MAX, capRate + 0.5); applyRate(); }
   601→   622→                  }
   602→   623→                }
   603→   624→              }
   604→   625→              lastM = m;
   605→   626→              doneCb && doneCb();
   606→   627→            }
   607→   628→            /* backpressure: pause bila encoder tertinggal, lanjut otomatis */
   608→   629→            if (venc.encodeQueueSize > CAP) { stop(); setTimeout(pump, 5); return; }
   609→   630→            pump();
   610→   631→          } catch (e) {
   611→   632→            refilling = false;
   612→   633→            fail(e && e.message ? e : new Error(String(e)));
   613→   634→          }
   614→   635→        };
   615→   636→        const pump = () => {
   616→   637→          if (finished) return;
   617→   638→          if (state.abort) return fail(new Error('Dibatalkan'));
   618→   639→          if (state.exportFatal) return fail(state.exportFatal);   /* v2.10 */
   619→   640→          if (v.ended || lastM >= seg.end - 0.004) return fin();
   620→   641→          if (v.paused) {
   621→   642→            try { const p = v.play(); if (p && p.catch) p.catch(err => fail(new Error('video tidak bisa diputar: ' + (err.message || err)))); }
   622→   643→            catch (err) { return fail(new Error('video tidak bisa diputar: ' + (err.message || err))); }
   623→   644→          }
   624→   645→          try { v.requestVideoFrameCallback(onFrame); } catch (e) { fin(); }
   625→   646→        };
   626→   647→        v.addEventListener('ended', fin, { once: true });
   627→   648→        v.addEventListener('error', () => fail(new Error('video error saat dirender')), { once: true });
   628→   649→        pump();
   629→   650→      });
   630→   651→      if (refillN) console.info(`part ${seg.i + 1}: ${n} frame dirender · ${refillN} diisi-ulang (lompatan maks ${Math.round(maxGapMs)} ms) — anti-beku v2.11 aktif`);
   631→   652→      if (n === 0) {
   632→   653→        /* sumber aneh / tidak menghasilkan frame — jalankan jalur lama */
   633→   654→        _stage = 'fallback-seek';
   634→   655→        await renderFramesBySeek(v, ecv, ec, venc, seg, fps, slot, doneCb, false, spd, mergeCtx ? mergeCtx.off : 0);
   635→   656→        n = slot.fr;
   636→   657→      }
   637→   658→    } else {
   638→   659→      /* audio-only / tanpa rVFC: jalur frame loop (tanpa seek utk audio) */
