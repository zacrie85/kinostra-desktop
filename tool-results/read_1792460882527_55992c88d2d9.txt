    83→function fmtClock(t) {
    84→  t = Math.max(0, Math.round(t || 0));
    85→  const h = Math.floor(t / 3600), m = Math.floor((t % 3600) / 60), s = t % 60;
    86→  return h ? `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
    87→           : `${m}:${String(s).padStart(2, '0')}`;
    88→}
    89→function syncRangeInputs() {

    93→  rs.value = fmtClock(r.start);

    99→    : `PENUH — 0:00 → ${fmtT(dur)} · <b>${segmentsCount()}</b> part`;
   100→}
   101→function setRangeStart(sec) {
   102→  if (!state.file) { toast('Impor media dulu', 'warn'); return; }
   103→  if (isNaN(sec)) { toast('Format waktu: detik (90) atau 1:30 / 1:02:03', 'warn'); return; }
   104→  const dur = state.duration || 0;
   105→  sec = clamp(sec, 0, Math.max(0, dur - 0.5));
   106→  if (sec > rangeInfo().end - 0.5) { toast('Waktu mulai terlalu dekat/melebihi waktu berhenti', 'warn'); return; }
   107→  state.rangeStart = sec; updateAll(); syncRangeInputs();
   108→}
   109→function setRangeEnd(sec) {
   110→  if (!state.file) { toast('Impor media dulu', 'warn'); return; }
   111→  if (isNaN(sec)) { toast('Kosongkan = sampai habis, atau isi 1:30 / 90', 'warn'); return; }
   112→  const dur = state.duration || 0;
   113→  sec = clamp(sec, 0, dur);
   114→  if (sec <= state.rangeStart + 0.5) { toast('Waktu berhenti harus sesudah waktu mulai', 'warn'); return; }
   115→  state.rangeEnd = (sec >= dur - 0.05) ? 0 : sec;   /* menempel akhir = sampai habis */
   116→  updateAll(); syncRangeInputs();
   117→}
