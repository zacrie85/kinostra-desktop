    49→async function probeEncoderFPS(cfg, W, H, frames = 30) {
    50→  let venc = null;
    51→  try {
    52→    const c = document.createElement('canvas'); c.width = W; c.height = H;
    53→    const g = c.getContext('2d');
    54→    g.fillStyle = '#1a2a3a'; g.fillRect(0, 0, W, H);
    55→    let chunks = 0, err = null, flushed = false;
    56→    venc = new VideoEncoder({
    57→      output: () => chunks++,
    58→      error: e => { err = err || (e && e.message) || 'encoder error'; }
    59→    });
    60→    venc.configure({ ...cfg, latencyMode: 'realtime' });
    61→    const t0 = performance.now();
    62→    for (let i = 0; i < frames; i++) {
    63→      /* konten BERGERAK — encoder bekerja nyata, angka fps jujur */
    64→      g.fillStyle = `hsl(${(i * 12) % 360} 60% ${18 + (i % 26)}%)`;
    65→      g.fillRect(0, 0, W, H);
    66→      g.fillStyle = '#fff';
    67→      g.fillRect((i * 37) % (W - 130), (i * 53) % (H - 70), 120, 68);
    68→      const vf = new VideoFrame(c, { timestamp: i * 41666, duration: 41666 });
    69→      venc.encode(vf, { keyFrame: i === 0 }); vf.close();
    70→    }
    71→    await withTimeout(venc.flush().then(() => { flushed = true; }), 10000, 'probe flush macet');
    72→    const ms = performance.now() - t0;
    73→    const ok = flushed && chunks >= frames * 0.9 && !err;
    74→    return { ok, fps: ok ? Math.round(frames / Math.max(1, ms) * 1000) : 0 };
    75→  } catch (e) {
    76→    return { ok: false, fps: 0 };
    77→  } finally {
    78→    /* v2.15 FIX BOCOR SESI ENCODER: dulu bila flush probe kena timeout,
    79→       exception melompati venc.close() → sesi encoder Media Foundation
    80→       TETAP TERBUKA di GPU process. Sesi bocor menumpuk sepanjang batch →
    81→       film ke-3 dst encoder hardware menolak/macet → tiap part membayar
    82→       gagal-hardware 30-45 dtk + render ulang software → ekspor 4× lipat.
    83→       Kini close() dijamin jalan di SEMUA jalur (sukses/gagal/timeout). */
    84→    try { if (venc && venc.state !== 'closed') venc.close(); } catch (e) { }
    85→  }
    86→}

   109→async function pickVCodecCfg(W, H, br, fps) {
   110→  const base = ['avc1.640028', 'avc1.4D0028', 'avc1.42002A'];
   111→  const mkHw = c => ({ codec: c, width: W, height: H, bitrate: br, framerate: fps, hardwareAcceleration: 'prefer-hardware' });
   112→  const mkSw = c => ({ codec: c, width: W, height: H, bitrate: br, framerate: fps, hardwareAcceleration: 'prefer-software' });
   113→  const key = `${W}x${H}:${br}:${fps}:${state.acc}`;
   114→  if (_accCache && _accCache.key === key) { _lastEnc = _accCache.res.enc; return _accCache.res.cfg; }
   115→
   116→  const probeKind = async (mk, kind) => {
   117→    for (const c of base) {
   118→      try {
   119→        const cfg = mk(c);
   120→        const s = await VideoEncoder.isConfigSupported(cfg);
   121→        if (!s.supported) continue;
   122→        const r = await probeEncoderFPS(cfg, W, H);
   123→        if (r.ok && r.fps > 0) return { cfg, enc: { kind, fps: r.fps } };
   124→      } catch (e) { }
   125→    }
   126→    return null;
   127→  };
   128→
   129→  let res = null;
   130→  if (state.acc === 'gpu') {
   131→    /* DIPAKSA GPU — bila encoder hardware tidak merespons, jatuh ke CPU
   132→       dengan peringatan (ekspor tetap jalan, bukan gagal) */
   133→    res = await probeKind(mkHw, 'hw');
   134→    if (!res) {
   135→      toast('Encoder GPU tidak merespons — otomatis memakai encoder CPU', 'warn');
   136→      res = await probeKind(mkSw, 'sw');
   137→    }
   138→  } else if (state.acc === 'cpu') {
   139→    res = (await probeKind(mkSw, 'sw')) || (await probeKind(mkHw, 'hw'));
   140→  } else {
   141→    /* OTOMATIS — ukur keduanya, pilih yang TERBUKTI lebih kencang.
   142→       GPU harus lebih cepat ≥15% untuk mengalahkan CPU (stabil & hemat risiko).
   143→       v2.15: bila hardware sudah didemosi (berulang kali macet di film
   144→       sebelumnya), langsung software — tanpa uji GPU lagi. */
   145→    const hw = (state.acc === 'auto' && _hwDemoted) ? null : await probeKind(mkHw, 'hw');
   146→    const sw = await probeKind(mkSw, 'sw');
   147→    res = !sw ? hw : (!hw ? sw : (hw.enc.fps >= sw.enc.fps * 1.15 ? hw : sw));
   148→  }
   149→  if (!res) {
   150→    /* upaya terakhir: tanpa preferensi hardware */
   151→    for (const c of base) {
   152→      try {
   153→        const cfg = { codec: c, width: W, height: H, bitrate: br, framerate: fps };
   154→        const s = await VideoEncoder.isConfigSupported(cfg);
   155→        const r = s.supported ? await probeEncoderFPS(cfg, W, H) : { ok: false };
   156→        if (r.ok) { res = { cfg, enc: { kind: 'sw', fps: r.fps } }; break; }
   157→      } catch (e) { }
   158→    }
   159→  }
   160→  if (res) {
   161→    _accCache = { key, res };
   162→    _lastEnc = res.enc;
   163→    return res.cfg;
   164→  }
   165→  return null;
   166→}

   230→    const s = oc.createBufferSource(); s.buffer = state.audioBuffer;
   231→    s.playbackRate.value = spd;
   232→    const g = oc.createGain(); g.gain.value = state.audioGain;
   233→    s.connect(g); g.connect(oc.destination);
   234→    s.start(0, start, Math.max(0.01, Math.min(durOut, (state.audioBuffer.duration - start) / spd)));
   235→  }
   236→  if (state.music.buffer && state.music.buffer.duration > 0.01) {
   237→    const s = oc.createBufferSource(); s.buffer = state.music.buffer;
   238→    s.playbackRate.value = spd;

   242→    s.start(0, start % state.music.buffer.duration, durOut);
   243→  }

   246→/* ---------- ENCODING AUDIO (v2.15: PER BLOK 60 dtk — RAM tetap datar) ----------
   247→   Dulu: SELURUH audio part dirender dulu oleh OfflineAudioContext →
   248→   AudioBuffer sebesar durasi part (film 60 mnt ≈ 700 MB Float32 di RAM
   249→   per part!) baru di-encode. Kini: blok 60 dtk dirender & di-encode
   250→   bergantian. Hasil identik — offset sumber & posisi loop musik kontinu
   251→   antar blok (blok k = waktu output [k·60, (k+1)·60) → sumber mulai

   253→async function encodeAudioStream(muxer, seg, durOut, spd = 1, codec) {
   254→  const sr = 48000, chs = 2;
   255→  const aenc = new AudioEncoder({ output: (c, m) => muxer.addAudioChunk(c, m), error: e => console.warn('aenc', e) });
   256→  aenc.configure({ codec, sampleRate: sr, numberOfChannels: chs, bitrate: 160000 });
   257→  try {
   258→    const BLK = 60;
   259→    let outOff = 0;
   260→    while (outOff < durOut - 1e-6) {
   261→      if (state.abort) throw new Error('Dibatalkan');
   262→      if (state.exportFatal) throw state.exportFatal;
   263→      const len = Math.min(BLK, durOut - outOff);
   264→      const mix = await renderMix(seg.start + outOff * spd, len, spd);
   265→      const L = mix.getChannelData(0), R2 = chs > 1 ? mix.getChannelData(1) : L;
   266→      for (let off = 0; off < mix.length; off += sr) {
   267→        if (state.abort) throw new Error('Dibatalkan');
   268→        if (state.exportFatal) throw state.exportFatal;
   269→        const n = Math.min(sr, mix.length - off);
   270→        const data = new Float32Array(n * chs);
   271→        data.set(L.subarray(off, off + n), 0);
   272→        if (chs > 1) data.set(R2.subarray(off, off + n), n);
   273→        const ad = new AudioData({ format: 'f32-planar', sampleRate: sr, numberOfFrames: n, numberOfChannels: chs,
   274→          timestamp: Math.round((outOff + off / sr) * 1e6), data });
   275→        aenc.encode(ad); ad.close();
   276→        let drainMs = 0;
   277→        while (aenc.encodeQueueSize > 10) { await sleep(2); if (state.abort) throw new Error('Dibatalkan'); if (state.exportFatal) throw state.exportFatal; if ((drainMs += 2) > 30000) throw new Error('encoder audio macet'); }
   278→      }
   279→      outOff += len;
   280→    }
   281→    await aenc.flush();
   282→  } finally {
   283→    try { aenc.close(); } catch (e) { }   /* v2.15: sesi encoder audio selalu ditutup */
   284→  }
   285→}

   318→async function renderFramesBySeek(v, ecv, ec, venc, seg, fps, slot, doneCb, noSeek = false, spd = 1) {
   319→  const n = Math.max(1, Math.round((seg.end - seg.start) / spd * fps));
   320→  let seekP = null;
   321→  for (let i = 0; i < n; i++) {
   322→    if (state.abort) throw new Error('Dibatalkan');
   323→    if (state.exportFatal) throw state.exportFatal;
   324→    if (seekP) { await seekP; seekP = null; }
   325→    const t = Math.min(seg.start + i * spd / fps, Math.max(0, (v.duration || seg.end) - 0.011));
   326→    setCompSrc(v);
   327→    drawComposition(ec, ecv.width, ecv.height, t);
   328→    setCompSrc(null);
   329→    const vf = new VideoFrame(ecv, { timestamp: Math.round(i * 1e6 / fps), duration: Math.round(1e6 / fps) });
   330→    venc.encode(vf, { keyFrame: i % (fps * 2) === 0 }); vf.close();
   331→    if (!noSeek && i + 1 < n) seekP = seekTo(seg.start + (i + 1) * spd / fps, v);
   332→    while (venc.encodeQueueSize > 14) await sleep(2);
   333→    slot.fr = i + 1;
   334→    doneCb && doneCb();
   335→  }
   336→  return n;
   337→}

   342→async function renderPartWorker(seg, srcUrl, W, H, fps, br, vCfg, aCodec, slot, doneCb) {
   343→  /* v2.12: KECEPATAN video — 1 = normal, 2 = 2× cepat (part ½ lebih pendek),
   344→     0.5 = slow-motion. Durasi output = durasi sumber ÷ spd. */

   349→  v.style.cssText = 'position:fixed;left:-9999px;top:-9999px;width:2px;height:2px;opacity:0';

   351→  document.body.appendChild(v);
   352→  v.addEventListener('error', () => console.error(`part ${seg.i + 1} media error: code=${v.error && v.error.code} msg=${v.error && v.error.message}`));
   353→  let venc = null, _stage = 'init', n = 0;
   354→  try {
   355→    await new Promise((res, rej) => {
   356→      const ok = () => { clean(); res(); };
   357→      const bad = () => { clean(); rej(new Error('worker: meta tidak termuat')); };
   358→      const clean = () => { v.removeEventListener('loadedmetadata', ok); v.removeEventListener('error', bad); };
   359→      v.addEventListener('loadedmetadata', ok); v.addEventListener('error', bad);
   360→      setTimeout(() => { if (v.readyState >= 1) ok(); }, 6000);
   361→    });
   362→    _stage = 'seek0';
   363→    await seekTo(seg.start, v);
   364→
   365→    _stage = 'canvas';
   366→    const ecv = document.createElement('canvas'); ecv.width = W; ecv.height = H;
   367→    const ec = ecv.getContext('2d', { alpha: false });
   368→    _stage = 'muxer';
   369→    /* v2.15: part BESAR ditulis LANGSUNG ke disk (StreamTarget → IPC 8 MB) —
   370→       dulu SELURUH berkas MP4 menumpuk di RAM (ArrayBufferTarget) sampai
   371→       render selesai; part 30-60 mnt = gigabyte per part di memori,
   372→       batch film panjang lalu kehabisan memori → ekspor merambat.
   373→       Part kecil tetap in-memory (fastStart penuh, kompatibilitas maks). */
   374→    const nm = partFileName(state.title, state.partPrefix, seg.i + 1);
   375→    const estBytes = Math.round(durS * (br + (aCodec ? 160000 : 0)) / 8) + 4e6;
   376→    const streamDisk = estBytes > (state._streamThreshMB || 384) * 1048576;
   377→    let wsId = null, wsErr = null, wsPath = nm, chain = Promise.resolve(), sent = 0;
   378→    const mkAudio = aCodec ? { audio: { codec: aCodec === 'mp4a.40.2' ? 'aac' : 'opus', numberOfChannels: 2, sampleRate: 48000 } } : {};
   379→    const mkVideo = { codec: 'avc', width: W, height: H, frameRate: fps };
   380→    let muxer;
   381→    if (streamDisk) {
   382→      const w = await window.kinostra.beginWrite(slot.dir, nm);
   383→      wsId = w.id; wsPath = w.path;
   384→      console.info(`part ${seg.i + 1}: muxer streaming disk aktif (±${fmtMB(estBytes)}) — RAM tetap datar (v2.15)`);
   385→      muxer = new Mp4Muxer.Muxer({
   386→        target: new Mp4Muxer.StreamTarget({
   387→          chunked: true, chunkSize: 8 * 1024 * 1024,
   388→          /* onData(data, posisi) — tulis-acak ke disk via pwrite (IPC);
   389→             rantai `chain` menjaga urutan panggilan, patch header mdat di
   390→             posisi lama oleh mp4-muxer (fastStart:false) tetap benar */
   391→          onData: (data, pos) => {
   392→            if (wsErr) return;
   393→            sent += data.byteLength;
   394→            slot.saveProg = Math.min(0.98, sent / estBytes);
   395→            chain = chain.then(() => window.kinostra.writeChunkAt(wsId, data, pos)).catch(e => { wsErr = wsErr || e; });
   396→          }
   397→        }),
   398→        fastStart: false, firstTimestampBehavior: 'offset',
   399→        video: mkVideo, ...mkAudio
   400→      });
   401→    } else {
   402→      muxer = new Mp4Muxer.Muxer({
   403→        target: new Mp4Muxer.ArrayBufferTarget(), fastStart: 'in-memory', firstTimestampBehavior: 'offset',
   404→        video: mkVideo, ...mkAudio
   405→      });
   406→    }
   407→    if (aCodec) {
   408→      _stage = 'audio';
   409→      /* v2.15: encode audio PER BLOK — RAM datar di part panjang */
   410→      await withTimeout(encodeAudioStream(muxer, seg, durS, spd, aCodec), 300000, 'encoding audio macet');
   411→    }
   412→    _stage = 'configure';
   413→    const hw = vCfg.hardwareAcceleration === 'prefer-hardware';
   414→    /* v2.7 FIX "SUARA JALAN, GAMBAR BEKU": encoder GPU Windows (Media
   415→       Foundation) dengan latencyMode 'quality' dapat mengeluarkan chunk
   416→       B-frame berurutan DECODE (timestamp bolak-balik). mp4-muxer melempar
   417→       error monotonic DI DALAM callback output → pengecualian tertelan
   418→       diam-diam → hanya beberapa frame pertama masuk MP4 → gambar beku +
   419→       audio normal + ekspor "berhasil". Perisai 4 lapis:
   420→       1) latencyMode 'realtime' utk SEMUA encoder (tanpa reorder B-frame)
   421→       2) timestamp turun > 5 ms terdeteksi → gagal cepat → auto-retry SW
   422→       3) error encoder diteruskan ke promise (bukan hanya console)
   423→       4) verifikasi jumlah chunk video setelah flush (≥ 90% frame) */
   424→    let vChunks = 0, lastVts = -Infinity, vencErrMsg = null;
   425→    venc = new VideoEncoder({
   426→      output: (c, m) => {
   427→        const ts = typeof c.timestamp === 'number' ? c.timestamp : 0;
   428→        if (vChunks > 0 && ts <= lastVts) {
   429→          if (lastVts - ts > 5000) {
   430→            vencErrMsg = `encoder mengeluarkan frame dengan urutan waktu tidak naik (${Math.round(lastVts / 1000)} → ${Math.round(ts / 1000)} ms)`;
   431→            console.error('venc-stream', vencErrMsg);
   432→          }
   433→          return; /* jangan masukkan chunk rusak/nol-durasi ke muxer */
   434→        }
   435→        lastVts = ts; vChunks++;
   436→        muxer.addVideoChunk(c, m);
   437→      },
   438→      error: e => { vencErrMsg = (e && e.message) || String(e); console.error('venc', vencErrMsg); }
   439→    });
   440→    venc.configure({ ...vCfg, latencyMode: 'realtime' });
   441→    const CAP = hw ? 40 : 14;
   442→
   443→    /* gambar 1 frame pada waktu media m → encode dengan timestamp nyata
   444→       v2.12: PTS = waktu OUTPUT = (m − mulai part) ÷ kecepatan */
   445→    const drawAt = (t, ts) => {
   446→      setCompSrc(v);
   447→      drawComposition(ec, W, H, t);
   448→      setCompSrc(null);
   449→      const vf = new VideoFrame(ecv, { timestamp: ts, duration: Math.round(1e6 / fps) });
   450→      venc.encode(vf, { keyFrame: n % (fps * 2) === 0 }); vf.close();
   451→      n++; slot.fr = n;
   452→    };
   453→    const outTs = (m) => Math.max(0, Math.round(((m - seg.start) / spd) * 1e6));
   454→
   455→    if (RVFC_OK && !state.isAudio) {
   456→      /* ============ v2.2: CAPTURE PLAYBACK (tanpa seek per frame) ============
   457→         v2.5: + WATCHDOG ANTI-MACET. Penyebab lama ekspor "macet di bagian
   458→         akhir tanpa pesan": frame callback (rVFC) berhenti memicu di frame
   459→         terakhir & event ended tidak sampai, atau encoder GPU berhenti
   460→         mengeluarkan output → promise render tidak pernah selesai.
   461→         Sekarang interval 250 ms memantau: video selesai? waktu media
   462→         sudah mencapai akhir part? video error? encoder diam terlalu
   463→         lama? → semua diarahkan SELESAI atau GAGAL DENGAN PESAN JELAS. */
   464→      _stage = 'stream';
   465→      /* v2.11 FIX "KADANG GERAK KADANG BEKU": dulu capRate mulai 2× dan bisa
   466→         naik sampai 4× — di atas kemampuan decoder/compositor, frame sumber
   467→         dibuang beramai-ramai, mediaTime antar frame tersaji LOMPAT, PTS
   468→         ekspor ikut bolong → pemutar menahan frame lama berdetik-detik.
   469→         Sekarang: mulai 1.5×, turun SEGERA saat ada frame hilang, dan setiap
   470→         lompatan DIISI ULANG dengan seek presisi (refillGap) sehingga aliran
   471→         PTS selalu rapat & mulus.
   472→         v2.14: bila encoder GPU (hardware) aktif, langit-langit dinaikkan
   473→         ke 3× — encoding bukan lagi hambatan, decoder-lah penentu. */
   474→      const RATE_MAX = hw ? 3 : (_xPar >= 3 ? 2 : 2.5);
   475→      /* v2.12: toleransi & target lompatan diskalakan kecepatan —
   476→         pada spd 2× jarak sumber antar frame tampil 2× lebih renggang */
   477→      const frameGapTol = 1.8 / fps * spd;
   478→      let capRate = 1.5, dAcc = 0, dN = 0, lastM = -1, hadFirst = false;
   479→      const applyRate = () => { try { v.playbackRate = Math.min(4.2, capRate * spd); } catch (e) { } };
   480→      applyRate();
   481→      let lastAct = Date.now(), qStallSince = 0;
   482→      let refilling = false, refillN = 0, maxGapMs = 0;
   483→      /* v2.11: isi ulang frame yang hilang di interval (from, to) — seek
   484→         presisi slot per slot, encode dengan PTS aslinya. Dipanggil SEBELUM
   485→         frame@to di-encode agar urutan PTS tetap naik. lastAct disegarkan
   486→         tiap slot supaya watchdog 30 dtk tidak mengira render diam. */
   487→      const refillGap = async (from, to) => {
   488→        const step = spd / fps;
   489→        let t = from + step, k = 0;
   490→        while (t < to - step * 0.5) {
   491→          if (state.abort) throw new Error('Dibatalkan');
   492→          if (state.exportFatal) throw state.exportFatal;
   493→          await seekTo(Math.min(t, Math.max(0, (v.duration || seg.end) - 0.011)), v);
   494→          drawAt(t, outTs(t));
   495→          lastAct = Date.now(); k++; t += step;
   496→          doneCb && doneCb();
   497→        }
   498→        refillN += k; return k;
   499→      };
   500→      await new Promise((resolve, rej) => {
   501→        let finished = false;
   502→        const stop = () => { try { v.pause(); } catch (e) { } };
   503→        const fin = () => { if (finished) return; finished = true; stop(); clearInterval(watch); resolve(); };
   504→        const fail = (err) => { if (finished) return; finished = true; stop(); clearInterval(watch); rej(err); };
   505→        /* --- v2.5: WATCHDOG (detak 250 ms) --- */
   506→        const watch = setInterval(() => {
   507→          if (finished) { clearInterval(watch); return; }
   508→          if (state.abort) return fail(new Error('Dibatalkan'));
   509→          if (state.exportFatal) return fail(state.exportFatal);   /* v2.10: part lain sudah gagal */
   510→          /* v2.7: error encoder terdeteksi → gagal cepat → retry software */
   511→          if (vencErrMsg) return fail(new Error('encoder video bermasalah: ' + vencErrMsg));
   512→          if (v.error) return fail(new Error('video rusak/tak terbaca: ' + (v.error.message || ('kode ' + v.error.code))));
   513→          /* KUNCI FIX: tangani akhir video walau frame callback mati —
   514→             inilah yang dulu bikin macet selamanya di bagian akhir */
   515→          if (v.ended || v.currentTime >= seg.end - 0.004) return fin();
   516→          const now = Date.now();
   517→          if (venc.encodeQueueSize > CAP) {
   518→            if (!qStallSince) qStallSince = now;
   519→            else if (now - qStallSince > 45000)
   520→              return fail(new Error('encoder video macet (antrean tidak mengalir 45 dtk) — akan dicoba ulang otomatis dengan encoder software'));
   521→          } else qStallSince = 0;
   522→          if (now - lastAct > 30000)
   523→            return fail(new Error('render diam total 30 dtk — akan dicoba ulang otomatis dengan encoder software'));
   524→          /* dorong ulang bila video diam padahal encoder tidak penuh */
   525→          if (v.paused && !refilling && !v.seeking && venc.encodeQueueSize <= CAP && !v.ended) {
   526→            try { const p = v.play(); if (p && p.catch) p.catch(err => fail(new Error('video tidak bisa dilanjut: ' + (err.message || err)))); }
   527→            catch (err) { fail(new Error('video tidak bisa dilanjut: ' + (err.message || err))); }
   528→          }
   529→        }, 250);
   530→        const onFrame = async (now, meta) => {
   531→          try {
   532→            if (finished) return;
   533→            lastAct = Date.now();
   534→            if (state.abort) return fail(new Error('Dibatalkan'));
   535→            if (state.exportFatal) return fail(state.exportFatal);   /* v2.10 */
   536→            const m = meta.mediaTime;
   537→            if (v.ended || m >= seg.end - 0.004) return fin();
   538→            if (m >= seg.start - 0.06 && m > lastM) {
   539→              /* v2.11: DETEKSI FRAME HILANG — lompatan mediaTime > 1.8× durasi
   540→                 frame berarti decoder/compositor membuang frame. Isi slot yang
   541→                 bolong lewat seek presisi SEBELUM frame m di-encode supaya
   542→                 urutan PTS rapat (tanpa durasi sampel multi-detik = beku). */
   543→              const gap = hadFirst ? m - lastM : 0;
   544→              if (gap > frameGapTol) {
   545→                maxGapMs = Math.max(maxGapMs, gap * 1000);
   546→                refilling = true; stop();
   547→                const k = await refillGap(lastM, m);
   548→                if (finished) return;
   549→                await seekTo(Math.min(m, Math.max(0, (v.duration || seg.end) - 0.011)), v);
   550→                refilling = false;
   551→                if (finished) return;
   552→                /* decoder kewalahan pada rate ini → turun SEGERA */
   553→                if (capRate > 1) { capRate = Math.max(1, capRate - 0.5); applyRate(); }
   554→                dAcc = 0; dN = 0;
   555→              }
   556→              drawAt(m, outTs(m));
   557→              hadFirst = true;
   558→              /* rate adaptif: naik HANYA bila jendela bersih tanpa lompatan
   559→                 v2.12: target jarak antar frame = spd/fps (waktu sumber) */
   560→              if (lastM >= 0) {
   561→                const d = m - lastM;
   562→                if (d > 0 && d < 0.6 * spd) {
   563→                  dAcc += d; dN++;
   564→                  if (dN >= 24) {
   565→                    const avg = dAcc / dN; dAcc = 0; dN = 0; const tg = spd / fps;
   566→                    if (avg > tg * 1.5 && capRate > 1) { capRate = Math.max(1, capRate - 0.5); applyRate(); }
   567→                    else if (avg < tg * 1.18 && capRate < RATE_MAX) { capRate = Math.min(RATE_MAX, capRate + 0.5); applyRate(); }
   568→                  }
   569→                }
   570→              }
   571→              lastM = m;
   572→              doneCb && doneCb();
   573→            }
   574→            /* backpressure: pause bila encoder tertinggal, lanjut otomatis */
   575→            if (venc.encodeQueueSize > CAP) { stop(); setTimeout(pump, 5); return; }
   576→            pump();
   577→          } catch (e) {
   578→            refilling = false;
   579→            fail(e && e.message ? e : new Error(String(e)));
   580→          }
   581→        };
   582→        const pump = () => {
   583→          if (finished) return;
   584→          if (state.abort) return fail(new Error('Dibatalkan'));
   585→          if (state.exportFatal) return fail(state.exportFatal);   /* v2.10 */
   586→          if (v.ended || lastM >= seg.end - 0.004) return fin();
   587→          if (v.paused) {
   588→            try { const p = v.play(); if (p && p.catch) p.catch(err => fail(new Error('video tidak bisa diputar: ' + (err.message || err)))); }
   589→            catch (err) { return fail(new Error('video tidak bisa diputar: ' + (err.message || err))); }
   590→          }
   591→          try { v.requestVideoFrameCallback(onFrame); } catch (e) { fin(); }
   592→        };
   593→        v.addEventListener('ended', fin, { once: true });
   594→        v.addEventListener('error', () => fail(new Error('video error saat dirender')), { once: true });
   595→        pump();
   596→      });
   597→      if (refillN) console.info(`part ${seg.i + 1}: ${n} frame dirender · ${refillN} diisi-ulang (lompatan maks ${Math.round(maxGapMs)} ms) — anti-beku v2.11 aktif`);
   598→      if (n === 0) {
   599→        /* sumber aneh / tidak menghasilkan frame — jalankan jalur lama */
   600→        _stage = 'fallback-seek';
   601→        await renderFramesBySeek(v, ecv, ec, venc, seg, fps, slot, doneCb, false, spd);
   602→        n = slot.fr;
   603→      }
   604→    } else {
   605→      /* audio-only / tanpa rVFC: jalur frame loop (tanpa seek utk audio) */
   606→      _stage = 'frames';
   607→      await renderFramesBySeek(v, ecv, ec, venc, seg, fps, slot, doneCb, !!state.isAudio, spd);
   608→      n = slot.fr;
   609→    }
   610→
   611→    _stage = 'flush';
   612→    slot.stage = 'flush';
   613→    /* v2.5: flush dengan batas waktu — encoder yang hang tidak lagi
   614→       membuat ekspor macet selamanya tanpa pesan */
   615→    await withTimeout(venc.flush(), 60000, 'encoder video tidak selesai (flush macet 60 dtk)');
   616→    venc.close(); venc = null;
   617→    /* v2.7: VERIFIKASI ALIRAN VIDEO — tangkap dua mode kegagalan encoder
   618→       GPU yang "diam": (a) error/reorder terdeteksi saat merender,
   619→       (b) output kurang dari 90% frame yang dirender. Tanpa ini MP4 tetap
   620→       tersimpan tapi isinya gambar beku. Gagal di sini → auto-retry dengan
   621→       encoder software oleh exportPartsToDir. */
   622→    if (vencErrMsg) throw new Error('aliran video tidak valid: ' + vencErrMsg);
   623→    /* v2.11: verifikasi diperketat 90% → 98% — encoder yang membuang output
   624→       di tengah stream kini terdeteksi → auto-retry software, bukan MP4 beku */
   625→    const minChunks = Math.max(1, Math.ceil(n * 0.98));
   626→    if (vChunks < minChunks)
   627→      throw new Error(`encoder hanya mengeluarkan ${vChunks} dari ${n} frame (gambar akan beku)`);
   628→    _stage = 'finalize';
   629→    slot.stage = 'finalize';
   630→    if (wsErr) throw wsErr;
   631→    await withTimeout(Promise.resolve().then(() => muxer.finalize()), 30000, 'penulisan header MP4 macet');
   632→    _stage = 'save';
   633→    slot.stage = 'save';
   634→    let wr;
   635→    if (streamDisk) {
   636→      /* v2.15: semua byte sudah mengalir ke disk SELAMA render — tinggal
   637→         menuntaskan rantai tulis, menutup stream, membaca ukuran berkas */
   638→      if (wsErr) throw wsErr;
   639→      await withTimeout(chain, 180000, 'menulis file ke disk macet (cek ruang kosong disk)');
   640→      if (wsId) { const id = wsId; wsId = null; await window.kinostra.endWrite(id); }
   641→      const stt = await window.kinostra.stat(wsPath);
   642→      wr = { path: wsPath, size: (stt && stt.ok) ? stt.size : sent };
   643→    } else {
   644→      wr = await withTimeout(
   645→        saveBufferToDir(muxer.target.buffer, slot.dir, nm, (w, t) => { slot.saveProg = w / t; }),
   646→        180000, 'menyimpan file ke disk macet (cek ruang kosong disk)');
   647→    }
   648→    slot.prog = 1; slot.fr = slot.total = Math.max(slot.total, n);
   649→    return { name: wr.path.split(/[\\/]/).pop(), path: wr.path, size: wr.size };
   650→  } catch (err) {
   651→    if (_stage !== 'frames' && _stage !== 'stream') console.error(`part ${seg.i + 1} GAGAL di tahap ${_stage}: ${err && err.name || 'ERR'} | ${err && err.message || err}`);
   652→    throw err;
   653→  } finally {
   654→    try { if (venc && venc.state !== 'closed') venc.close(); } catch (e) { }
   655→    /* v2.15: bila gagal di tengah render streaming → buang berkas setengah jadi */
   656→    try { if (wsId) { const id = wsId; wsId = null; window.kinostra.abortWrite(id).catch(() => { }); } } catch (e) { }
   657→    try { v.pause(); v.removeAttribute('src'); v.load(); } catch (e) { }
   658→    try { v.remove(); } catch (e) { }
   659→  }
   660→}

   662→/* ---------- INTI RENDER: SEMUA PART PARALEL -> folder dir ----------
   663→   ui: {title(segs), prog(p), sub(s), canceled()} */
   664→async function exportPartsToDir(dir, ui) {
   665→  await ensureFontsReady();
   666→  /* v2.6: pastikan decode audio asli (yang kini berjalan di belakang saat
   667→     loading) sudah selesai sebelum menentukan ada tidaknya track audio */
   668→  if (state.audioReady) { try { await state.audioReady; } catch (e) { } }
   669→  const fps = state.fps, sc = parseFloat(state.scale), q = QUAL[state.quality];
   670→  /* v2.12: kecepatan video global — durasi output = sumber ÷ spd */
   671→  const spd = clamp(state.speed || 1, 0.25, 4);
   672→  const dims = outDims(sc), { W, H } = dims;
   673→  const br = Math.round(q.br * (sc === 1 ? 1 : sc === 0.75 ? 0.62 : 0.38));
   674→  /* v2.14: pilih encoder sesuai AKSELERASI (auto = uji kecepatan GPU vs CPU) */
   675→  const vCfg = await pickVCodecCfg(W, H, br, fps);
   676→  if (!vCfg) throw new Error('Tidak ada codec video yang didukung engine');
   677→  const enc = _lastEnc;
   678→  /* v2.5: cadangan encoder SOFTWARE untuk percobaan ulang otomatis */
   679→  const swCfg = await pickSWCfg(W, H, br, fps);
   680→  const wantAudio = !!(state.audioBuffer || state.music.buffer);
   681→  const aCodec = wantAudio ? await pickACodec() : null;
   682→  if (wantAudio && !aCodec) toast('Encoder audio tak tersedia — ekspor tanpa audio', 'warn');
   683→  const segs = segments();
   684→  const par = pickParallelCount(segs.length, enc.kind === 'hw');
   685→  _xPar = par;   /* v2.11: langit-langit playbackRate capture menyesuaikan beban paralel */
   686→  const hw = vCfg.hardwareAcceleration === 'prefer-hardware';
   687→
   688→  /* slot progres per part — total frame = durasi output × fps */
   689→  const slots = segs.map(s => ({ seg: s, dir, prog: 0, fr: 0, total: Math.max(1, Math.round((s.end - s.start) / spd * fps)), working: false, stage: null, saveProg: 0 }));
   690→  const totalFr = slots.reduce((a, s) => a + s.total, 0);
   691→  const t0 = performance.now();
   692→  let lastUi = 0;
   693→  const uiTick = (force) => {
   694→    const now = performance.now();
   695→    if (!force && now - lastUi < 120) return;
   696→    lastUi = now;
   697→    const done = slots.reduce((a, s) => a + Math.min(s.fr, s.total), 0);
   698→    if (ui.prog) ui.prog(done / totalFr);
   699→    if (ui.sub) {
   700→      const el = Math.max(0.001, (now - t0) / 1000);
   701→      const rt = (done / fps / el);
   702→      const parts = slots.map(s => {
   703→        const pct = Math.round(100 * Math.min(s.fr, s.total) / s.total);
   704→        const stg = s.stage === 'flush' ? '· FLUSH' : s.stage === 'finalize' ? '· FINAL' : s.stage === 'save' ? '· SIMPAN' : '';
   705→        const tag = s.retried && s.working ? ' (ULANGI-SW)' : '';
   706→        return s.working ? `P${String(s.seg.i + 1).padStart(2, '0')} ${pct}%${tag}${stg}` : null;
   707→      }).filter(Boolean).join(' · ');
   708→      ui.sub(`${parts || 'menunggu…'}${rt > 0.05 ? ` · ${rt.toFixed(1)}× realtime` : ''}`);
   709→    }
   710→  };
   711→
   712→  /* v2.6: file virtual (impor/batch streaming) — worker memutar langsung
   713→     dari disk via kfile:// (Range), tanpa menyalin file ke memori */
   714→  const srcUrl = state.file.virtual ? state.file.kurl : URL.createObjectURL(state.file);
   715→  const results = new Array(segs.length).fill(null);
   716→  state.exportFatal = null;   /* v2.10: mulai ekspor baru — bersihkan sisa gagal lama */
   717→  try {
   718→    if (ui.title) ui.title(segs);
   719→    if (ui.prog) ui.prog(0);
   720→    if (ui.sub) ui.sub(`encoder ${enc.kind === 'hw' ? `GPU ${enc.fps}fps` : `CPU ${enc.fps}fps`} · ${par} render paralel${RVFC_OK && !state.isAudio ? ' · turbo stream' : ''}${spd !== 1 ? ` · kecepatan ${spd}×` : ''}…`);
   721→
   722→    /* jalankan worker dengan batas `par` sekaligus (antrean)
   723→       v2.10: kegagalan part TIDAK lagi men-set state.abort (flag batal
   724→       milik pengguna) — cukup state.exportFatal supaya worker lain berhenti.
   725→       Penelepon (ekspor tunggal / batch) menerima Error aslinya dan bisa
   726→       menampilkan penyebabnya; batch menandai video GAGAL lalu LANJUT. */
   727→    let cursor = 0;
   728→    const launch = async () => {
   729→      while (cursor < slots.length && !xAborted()) {
   730→        const idx = cursor++;
   731→        const slot = slots[idx];
   732→        slot.working = true; uiTick(true);
   733→        try {
   734→          results[idx] = await renderPartWorker(slot.seg, srcUrl, W, H, fps, br, vCfg, aCodec, slot, () => uiTick(false));
   735→        } catch (e) {
   736→          /* v2.5: RETRY OTOMATIS 1× dengan encoder SOFTWARE — penyembuh
   737→             utama "ekspor macet di bagian akhir" (encoder GPU berhenti).
   738→             Jangan ulangi bila user membatalkan sendiri / part lain gagal.
   739→             v2.15: kegagalan jalur hardware dicatat — 2× jatuh → sisa batch
   740→             otomatis memakai software (anti-pelan, tanpa dobel kerja). */
   741→          if (hw && !xAborted()) noteHwFailure(e && e.message || String(e));
   742→          if (!xAborted() && swCfg) {
   743→            console.warn(`part ${idx + 1} gagal (${e && e.message || e}) — ulangi 1× dengan encoder software`);
   744→            slot.fr = 0; slot.stage = null; slot.saveProg = 0; slot.retried = true; slot.working = true; uiTick(true);
   745→            try {
   746→              results[idx] = await renderPartWorker(slot.seg, srcUrl, W, H, fps, br, swCfg, aCodec, slot, () => uiTick(false));
   747→              console.info(`part ${idx + 1} sukses pada percobaan ke-2 (encoder software)`);
   748→            } catch (e2) {
   749→              const det2 = e2 && (e2.name + ' | ' + e2.message) || String(e2);
   750→              console.error(`part ${idx + 1} GAGAL juga di percobaan ke-2: ${det2}`);
   751→              if (!xAborted()) state.exportFatal = e2;
   752→              slot.working = false; uiTick(true);
   753→              return;
   754→            }
   755→          } else {
   756→            const det = e && (e.name + ' | ' + e.message + ' | ' + String(e.stack || '').split('\n')[1] || '') || String(e);
   757→            console.error(`part ${idx + 1} GAGAL: ${det}`);
   758→            if (!xAborted()) state.exportFatal = e;
   759→            slot.working = false; uiTick(true);
   760→            return;
   761→          }
   762→        }
   763→        slot.working = false; uiTick(true);
   764→      }
   765→    };
   766→    const runners = [];
   767→    for (let k = 0; k < par; k++) runners.push(launch());
   768→    /* pantau progres selagi worker jalan
   769→       v2.5: + WATCHDOG GLOBAL — bila TIDAK ADA kemajuan apa pun dari semua
   770→       worker selama 45 dtk, hentikan dengan pesan jelas (bukan macam) */
   771→    await new Promise(res => {
   772→      let lastSig = '', lastSigAt = Date.now();
   773→      const iv = setInterval(() => {
   774→        uiTick(false);
   775→        if (xAborted() || results.every(r => r)) { clearInterval(iv); res(); return; }
   776→        const sig = slots.map(s => `${s.fr}:${s.working ? 1 : 0}:${s.stage || ''}:${Math.round((s.saveProg || 0) * 100)}`).join('|');
   777→        const now = Date.now();
   778→        if (sig !== lastSig) { lastSig = sig; lastSigAt = now; }
   779→        else if (now - lastSigAt > 45000) {
   780→          clearInterval(iv);
   781→          /* v2.10: watchdog = kegagalan internal, BUKAN batal pengguna */
   782→          if (!xAborted()) {
   783→            state.exportFatal = new Error('Render berhenti merespons 45 detik — dihentikan otomatis. Coba set PARALEL 1× atau kualitas lebih rendah lalu ulangi.');
   784→          }
   785→          res();
   786→        }
   787→      }, 150);
   788→    });
   789→    await Promise.all(runners).catch(() => { });
   790→    if (state.exportFatal) { const err = state.exportFatal; throw err; }
   791→    if (state.abort && !results.every(r => r)) {
   792→      const made = results.filter(Boolean);
   793→      throw new Error(made.length ? `Dibatalkan — ${made.length} part tersimpan` : 'Dibatalkan');
   794→    }
   795→    if (results.some(r => !r)) throw new Error('Sebagian part gagal dirender');
   796→    if (ui.sub) {
   797→      const el = (performance.now() - t0) / 1000;
   798→      ui.sub(`selesai dalam ${el < 90 ? el.toFixed(0) + ' detik' : (el / 60).toFixed(1) + ' menit'} · ${par}× paralel${hw ? ' · GPU' : ''}`);
   799→    }
   800→    if (ui.prog) ui.prog(1);
   801→    return results;
   802→  } finally {
   803→    state.exportFatal = null;   /* v2.10: jangan bocor ke ekspor/video berikutnya */
   804→    if (!state.file.virtual) URL.revokeObjectURL(srcUrl);
   805→  }
   806→}

   809→async function doExport() {
   810→  if (!state.file) { toast('Impor media dulu', 'err'); return; }
   811→  if (state.busy) return;
   812→  if (!('VideoEncoder' in window)) { toast('Engine tidak mendukung WebCodecs', 'err'); return; }
   813→  state.busy = true; state.abort = false;
   814→  try {
   815→    const dir = await window.kinostra.pickOutputDir(_lastOutDir || undefined);
   816→    if (!dir) { state.busy = false; return; }
   817→    _lastOutDir = dir;
   818→    /* v2.13: AUTO-SUBTITEL — bila pengaturan aktif & video belum punya
   819→       subtitel, buatkan dulu di sini sehingga EKSPOR & SPLIT langsung
   820→       menghasilkan video bersubtitel tanpa langkah manual. Gagal internal
   821→       tidak membatalkan ekspor (toast peringatan + lanjut tanpa subtitel);
   822→       pembatalan pengguna dilempar sebagai 'Dibatalkan'. */
   823→    await ensureAutoSubs({ ownModal: true });
   824→    const tStart = performance.now();
   825→    const results = await exportPartsToDir(dir, {
   826→      title(segs) {
   827→        const par = pickParallelCount(segs.length, _lastEnc.kind === 'hw');
   828→        showModal({ title: `MERENDER ${segs.length} PART · PARALEL ${par}×`,
   829→          sub: 'Menyiapkan encoder…', cancel: true, onCancel() { state.abort = true; } });
   830→      },
   831→      prog: p => setProg(p),
   832→      sub: s => setSub(s)
   833→    });
   834→    const el = (performance.now() - tStart) / 1000;
   835→    hideModal(); showResults(results, dir);
   836→    toast(`${results.length} file selesai · ${el < 90 ? el.toFixed(0) + ' detik' : (el / 60).toFixed(1) + ' menit'}`, 'ok');
   837→  } catch (e) {
   838→    hideModal();
   839→    /* v2.10: hanya tampil "dibatalkan" bila BENAR-BENAR dibatalkan pengguna.
   840→       Kegagalan internal (encoder, media, watchdog) kini menampilkan
   841→       PENYEBAB aslinya — dulu semuanya tertelan jadi "Ekspor dibatalkan". */
   842→    if (state.abort || String(e.message || e).includes('Dibatalkan')) toast('Ekspor dibatalkan', 'warn');
   843→    else { console.error(e); toast('Ekspor gagal: ' + (e.message || e), 'err'); }
   844→  }
   845→  state.busy = false;
   846→}

   864→function updateOutName() {
   865→  if (!state.file) { $('#outName').textContent = '—'; return; }
   866→  const n = segmentsCount(), r = rangeInfo();

   875→function updateExportInfo() {
   876→  if (!state.file) { $('#expSum').textContent = 'Impor media untuk melihat ringkasan.'; return; }
   877→  const sc = parseFloat(state.scale), q = QUAL[state.quality];
   878→  const br = Math.round(q.br * (sc === 1 ? 1 : sc === 0.75 ? 0.62 : 0.38));
   879→  const dims = outDims(sc);
   880→  /* v2.12: estimasi memakai durasi efektif (rentang − potongan) ÷ kecepatan */
   881→  const outLen = expLen() / clamp(state.speed || 1, 0.25, 4);
   882→  const est = br / 8 * outLen + 160000 / 8 * outLen;
   883→  /* v2.14: paralel ikut mode akselerasi aktif */
   884→  const par = pickParallelCount(segmentsCount(), _lastEnc.kind === 'hw');
   885→  const r = rangeInfo();
   886→  const rangeLine = (r.start > 0.01 || (state.rangeEnd > 0 && r.end < (state.duration || 0) - 0.01))
   887→    ? `<br><b>RANGE ${fmtT(r.start)} → ${fmtT(r.end)}</b> · durasi ekspor ${fmtT(expLen())}` : '';
   888→  const cutLine = (state.cuts || []).length
   889→    ? ` · <b>POTONG ${state.cuts.length}</b> bagian (−${fmtT(cutsLen())})` : '';

   891→  $('#expSum').innerHTML = `<b>${segmentsCount()} PART</b> · ${dims.W}×${dims.H} · ${state.fps}fps · ${q.label} · PARALEL ${par}×${rangeLine}${cutLine}${spdLine}
