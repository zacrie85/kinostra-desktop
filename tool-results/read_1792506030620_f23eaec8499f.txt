    49→async function probeEncoderFPS(cfg, W, H, frames = 30) {
    50→  let venc = null;
    51→  try {
    52→    const c = document.createElement('canvas'); c.width = W; c.height = H;
    53→    const g = c.getContext('2d');
    54→    g.fillStyle = '#1a2a3a'; g.fillRect(0, 0, W, H);
    55→    let chunks = 0, err = null, flushed = false;
    56→    venc = new VideoEncoder({
    57→      output: () => chunks++,
    58→      error: e => { err = err || (e && e.message) || 'encoder error'; }
    59→    });
    60→    venc.configure({ ...cfg, latencyMode: 'realtime' });
    61→    const t0 = performance.now();
    62→    for (let i = 0; i < frames; i++) {
    63→      /* konten BERGERAK — encoder bekerja nyata, angka fps jujur */
    64→      g.fillStyle = `hsl(${(i * 12) % 360} 60% ${18 + (i % 26)}%)`;
    65→      g.fillRect(0, 0, W, H);
    66→      g.fillStyle = '#fff';
    67→      g.fillRect((i * 37) % (W - 130), (i * 53) % (H - 70), 120, 68);
    68→      const vf = new VideoFrame(c, { timestamp: i * 41666, duration: 41666 });
    69→      venc.encode(vf, { keyFrame: i === 0 }); vf.close();
    70→    }
    71→    await withTimeout(venc.flush().then(() => { flushed = true; }), 10000, 'probe flush macet');
    72→    const ms = performance.now() - t0;
    73→    const ok = flushed && chunks >= frames * 0.9 && !err;
    74→    return { ok, fps: ok ? Math.round(frames / Math.max(1, ms) * 1000) : 0 };
    75→  } catch (e) {
    76→    return { ok: false, fps: 0 };
    77→  } finally {
    78→    /* v2.15 FIX BOCOR SESI ENCODER: dulu bila flush probe kena timeout,
    79→       exception melompati venc.close() → sesi encoder Media Foundation
    80→       TETAP TERBUKA di GPU process. Sesi bocor menumpuk sepanjang batch →
    81→       film ke-3 dst encoder hardware menolak/macet → tiap part membayar
    82→       gagal-hardware 30-45 dtk + render ulang software → ekspor 4× lipat.
    83→       Kini close() dijamin jalan di SEMUA jalur (sukses/gagal/timeout). */
    84→    try { if (venc && venc.state !== 'closed') venc.close(); } catch (e) { }
    85→  }
    86→}

   109→async function pickVCodecCfg(W, H, br, fps) {
   110→  const base = ['avc1.640028', 'avc1.4D0028', 'avc1.42002A'];
   111→  const mkHw = c => ({ codec: c, width: W, height: H, bitrate: br, framerate: fps, hardwareAcceleration: 'prefer-hardware' });
   112→  const mkSw = c => ({ codec: c, width: W, height: H, bitrate: br, framerate: fps, hardwareAcceleration: 'prefer-software' });
   113→  const key = `${W}x${H}:${br}:${fps}:${state.acc}`;
   114→  if (_accCache && _accCache.key === key) { _lastEnc = _accCache.res.enc; return _accCache.res.cfg; }
   115→
   116→  const probeKind = async (mk, kind) => {
   117→    for (const c of base) {
   118→      try {
   119→        const cfg = mk(c);
   120→        const s = await VideoEncoder.isConfigSupported(cfg);
   121→        if (!s.supported) continue;
   122→        const r = await probeEncoderFPS(cfg, W, H);
   123→        if (r.ok && r.fps > 0) return { cfg, enc: { kind, fps: r.fps } };
   124→      } catch (e) { }
   125→    }
   126→    return null;
   127→  };
   128→
   129→  let res = null;
   130→  if (state.acc === 'gpu') {
   131→    /* DIPAKSA GPU — bila encoder hardware tidak merespons, jatuh ke CPU
   132→       dengan peringatan (ekspor tetap jalan, bukan gagal) */
   133→    res = await probeKind(mkHw, 'hw');
   134→    if (!res) {
   135→      toast('Encoder GPU tidak merespons — otomatis memakai encoder CPU', 'warn');
   136→      res = await probeKind(mkSw, 'sw');
   137→    }
   138→  } else if (state.acc === 'cpu') {
   139→    res = (await probeKind(mkSw, 'sw')) || (await probeKind(mkHw, 'hw'));
   140→  } else {
   141→    /* OTOMATIS — ukur keduanya, pilih yang TERBUKTI lebih kencang.
   142→       GPU harus lebih cepat ≥15% untuk mengalahkan CPU (stabil & hemat risiko).
   143→       v2.15: bila hardware sudah didemosi (berulang kali macet di film
   144→       sebelumnya), langsung software — tanpa uji GPU lagi. */
   145→    const hw = (state.acc === 'auto' && _hwDemoted) ? null : await probeKind(mkHw, 'hw');
   146→    const sw = await probeKind(mkSw, 'sw');
   147→    res = !sw ? hw : (!hw ? sw : (hw.enc.fps >= sw.enc.fps * 1.15 ? hw : sw));
   148→  }
   149→  if (!res) {
   150→    /* upaya terakhir: tanpa preferensi hardware */
   151→    for (const c of base) {
   152→      try {
   153→        const cfg = { codec: c, width: W, height: H, bitrate: br, framerate: fps };
   154→        const s = await VideoEncoder.isConfigSupported(cfg);
   155→        const r = s.supported ? await probeEncoderFPS(cfg, W, H) : { ok: false };
   156→        if (r.ok) { res = { cfg, enc: { kind: 'sw', fps: r.fps } }; break; }
   157→      } catch (e) { }
   158→    }
   159→  }
   160→  if (res) {
   161→    _accCache = { key, res };
   162→    _lastEnc = res.enc;
   163→    return res.cfg;
   164→  }
   165→  return null;
   166→}

   210→function withTimeout(p, ms, msg) {
   211→  return new Promise((res, rej) => {
   212→    const tm = setTimeout(() => rej(new Error(msg)), ms);
   213→    Promise.resolve(p).then(
   214→      v => { clearTimeout(tm); res(v); },
   215→      e => { clearTimeout(tm); rej(e); });
   216→  });

   218→async function renderMix(start, durOut, spd = 1) {
   219→  const sr = 48000, len = Math.max(1, Math.ceil(durOut * sr));
   220→  const oc = new OfflineAudioContext(2, len, sr);
   221→  /* v3.0: AUDIO SEHAT — rantai perbaikan: filter desis (opsional) →
   222→     normalizer (RMS diukur sekali per video) → kompresor lembut.
   223→     Bila mati, sumber tersambung langsung ke destination (dulu). */
   224→  let sink = null;
   225→  if (state.audioFix && state.audioFix.on) {
   226→    sink = oc.createGain();
   227→    let cur = sink;
   228→    if (state.audioFix.hiss) {
   229→      const hp = oc.createBiquadFilter(); hp.type = 'highpass'; hp.frequency.value = 95; hp.Q.value = 0.6;
   230→      const lp = oc.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 11500;
   231→      cur.connect(hp); hp.connect(lp); cur = lp;
   232→    }
   233→    const norm = oc.createGain();
   234→    const rms = (typeof measureAudioRms === 'function') ? measureAudioRms() : 0;
   235→    norm.gain.value = rms > 0.004 ? clamp(0.16 / rms, 0.5, 4) : 1;
   236→    cur.connect(norm);
   237→    const comp = oc.createDynamicsCompressor();
   238→    comp.threshold.value = -18; comp.knee.value = 20; comp.ratio.value = 2.6;
   239→    comp.attack.value = 0.01; comp.release.value = 0.24;
   240→    norm.connect(comp); comp.connect(oc.destination);
   241→  }
   242→  const connectSrc = node => { if (sink) node.connect(sink); else node.connect(oc.destination); };
   243→  /* v2.10 FIX: dulu s.start(0, start, durBuffer − start) melempar
   244→     InvalidStateError ketika start > durasi buffer (mis. WAKTU MULAI 1:05
   245→     padahal skor musik masih milik video sebelumnya yang lebih pendek) —
   246→     inilah pemicu utama batch ekspor "selalu dibatalkan" sejak v2.9.
   247→     Sekarang: audio asli dilewati bila sudah lewat, musik diputar BERULANG
   248→     (loop) sehingga durasi buffer berapa pun tidak pernah crash.
   249→     v2.12: kecepatan → audio asli diputar lebih cepat/lambat memakai
   250→     playbackRate (pitch ikut naik/turun seperti efek fast-forward klasik). */
   251→  if (state.audioBuffer && state.audioBuffer.duration > 0.01 && start < state.audioBuffer.duration - 0.01) {
   252→    const s = oc.createBufferSource(); s.buffer = state.audioBuffer;
   253→    s.playbackRate.value = spd;
   254→    const g = oc.createGain(); g.gain.value = state.audioGain;
   255→    s.connect(g); connectSrc(g);
   256→    s.start(0, start, Math.max(0.01, Math.min(durOut, (state.audioBuffer.duration - start) / spd)));
   257→  }
   258→  if (state.music.buffer && state.music.buffer.duration > 0.01) {
   259→    const s = oc.createBufferSource(); s.buffer = state.music.buffer;
   260→    s.playbackRate.value = spd;
   261→    const g = oc.createGain(); g.gain.value = state.music.gain;
   262→    s.connect(g); connectSrc(g);
   263→    s.loop = true;   /* buffer lebih pendek dari part → mengulang, bukan crash */
   264→    s.start(0, start % state.music.buffer.duration, durOut);
   265→  }
   266→  /* v3.3: BACKSOUND MP3 SENDIRI — file audio pilihan user, loop, volume
   267→     manual (state.bsound.gain). Offset mengikuti waktu media sehingga
   268→     posisi lagu KONTINU antar part/blok (sama dgn perilaku skor musik). */
   269→  const BS = state.bsound;
   270→  if (BS && BS.on && BS.buffer && BS.buffer.duration > 0.01) {
   271→    const s = oc.createBufferSource(); s.buffer = BS.buffer;
   272→    s.playbackRate.value = spd;
   273→    const g = oc.createGain(); g.gain.value = clamp(BS.gain, 0, 1.6);
   274→    s.connect(g); connectSrc(g);
   275→    s.loop = true;
   276→    s.start(0, start % BS.buffer.duration, durOut);
   277→  }
   278→  /* v3.0: TTS NARASI — suara AI offline diletakkan di waktu media
   279→     state.tts.at; ikut kecepatan video (spd). Potongan di luar part
   280→     dilewati; bagian yang terpotong di awal part disesuaikan. */
   281→  const T = state.tts;
   282→  if (T && T.buffer && T.buffer.duration > 0.01) {
   283→    let outOffN = (T.at - start) / spd, srcOff = 0;
   284→    if (outOffN < 0) { srcOff = -outOffN * spd; outOffN = 0; }
   285→    if (outOffN < durOut - 0.01 && srcOff < T.buffer.duration) {
   286→      const s = oc.createBufferSource(); s.buffer = T.buffer;
   287→      s.playbackRate.value = spd;
   288→      const g = oc.createGain(); g.gain.value = clamp(T.gain == null ? 0.9 : T.gain, 0, 1.6);
   289→      s.connect(g); connectSrc(g);
   290→      s.start(outOffN, srcOff, Math.max(0.01, Math.min(durOut - outOffN, (T.buffer.duration - srcOff) / spd)));
   291→    }
   292→  }
   293→  return oc.startRendering();
   294→}
   295→/* ---------- ENCODING AUDIO (v2.15: PER BLOK 60 dtk — RAM tetap datar) ----------
   296→   Dulu: SELURUH audio part dirender dulu oleh OfflineAudioContext →
   297→   AudioBuffer sebesar durasi part (film 60 mnt ≈ 700 MB Float32 di RAM
   298→   per part!) baru di-encode. Kini: blok 60 dtk dirender & di-encode
   299→   bergantian. Hasil identik — offset sumber & posisi loop musik kontinu
   300→   antar blok (blok k = waktu output [k·60, (k+1)·60) → sumber mulai
   301→   seg.start + outOff·spd), stempel waktu dihitung dari penghitung global.
   302→   v3.0: mergeOff = offset waktu output global saat GABUNG VIDEO;
   303→   fbuf dipakai ulang antar detik (kurangi tekanan GC ±40%). */
   304→async function encodeAudioStream(muxer, seg, durOut, spd = 1, codec, mergeOff = 0) {
   305→  const sr = 48000, chs = 2;
   306→  const aenc = new AudioEncoder({ output: (c, m) => muxer.addAudioChunk(c, m), error: e => console.warn('aenc', e) });
   307→  aenc.configure({ codec, sampleRate: sr, numberOfChannels: chs, bitrate: 160000 });
   308→  try {
   309→    const BLK = 60;
   310→    const fbuf = new Float32Array(sr * chs);
   311→    let outOff = 0;
   312→    while (outOff < durOut - 1e-6) {
   313→      if (state.abort) throw new Error('Dibatalkan');
   314→      if (state.exportFatal) throw state.exportFatal;
   315→      const len = Math.min(BLK, durOut - outOff);
   316→      const mix = await renderMix(seg.start + outOff * spd, len, spd);
   317→      const L = mix.getChannelData(0), R2 = chs > 1 ? mix.getChannelData(1) : L;
   318→      for (let off = 0; off < mix.length; off += sr) {
   319→        if (state.abort) throw new Error('Dibatalkan');
   320→        if (state.exportFatal) throw state.exportFatal;
   321→        const n = Math.min(sr, mix.length - off);
   322→        fbuf.set(L.subarray(off, off + n), 0);
   323→        if (chs > 1) fbuf.set(R2.subarray(off, off + n), n);
   324→        const ad = new AudioData({ format: 'f32-planar', sampleRate: sr, numberOfFrames: n, numberOfChannels: chs,
   325→          timestamp: Math.round((mergeOff + outOff + off / sr) * 1e6), data: fbuf });
   326→        aenc.encode(ad); ad.close();
   327→        let drainMs = 0;
   328→        while (aenc.encodeQueueSize > 10) { await sleep(2); if (state.abort) throw new Error('Dibatalkan'); if (state.exportFatal) throw state.exportFatal; if ((drainMs += 2) > 30000) throw new Error('encoder audio macet'); }
   329→      }
   330→      outOff += len;
   331→    }
   332→    await aenc.flush();
   333→  } finally {
   334→    try { aenc.close(); } catch (e) { }   /* v2.15: sesi encoder audio selalu ditutup */
   335→  }
   336→}

   349→/* ---------- JUMLAH RENDER PARALEL ----------

   352→function pickParallelCount(parts, hwOk = false) {
   353→  if (state.parallel > 0) return Math.max(1, Math.min(state.parallel, parts));
   354→  const cores = navigator.hardwareConcurrency || 8;
   355→  /* v2.15: GPU TERINTEGRASI (Intel Iris/UHD, AMD APU) — memori bersama
   356→     dgn RAM; paralel 6× membanjiri iGPU (encode+decode 6 sesi) → ekspor
   357→     justru merambat. Di iGPU langit-langitnya 4×. */
   358→  if (hwOk && state._igpu) return Math.max(2, Math.min(4, parts, Math.floor(cores / 2)));
   359→  return hwOk
   360→    ? Math.max(2, Math.min(6, parts, Math.floor(cores / 2)))
   361→    : Math.max(1, Math.min(4, parts, Math.floor(cores / 3)));
   362→}

   367→   v2.12: spd = kecepatan — frame ke-i = waktu output i/fps → sumber
   368→   i*spd/fps; PTS output tetap i*1e6/fps (monoton rapat). */
   369→async function renderFramesBySeek(v, ecv, ec, venc, seg, fps, slot, doneCb, noSeek = false, spd = 1, tsOff = 0) {
   370→  const n = Math.max(1, Math.round((seg.end - seg.start) / spd * fps));
   371→  let seekP = null;
   372→  for (let i = 0; i < n; i++) {
   373→    if (state.abort) throw new Error('Dibatalkan');
   374→    if (state.exportFatal) throw state.exportFatal;
   375→    if (seekP) { await seekP; seekP = null; }
   376→    const t = Math.min(seg.start + i * spd / fps, Math.max(0, (v.duration || seg.end) - 0.011));
   377→    setCompSrc(v);
   378→    drawComposition(ec, ecv.width, ecv.height, t);
   379→    setCompSrc(null);
   380→    const vf = new VideoFrame(ecv, { timestamp: Math.round(tsOff + i * 1e6 / fps), duration: Math.round(1e6 / fps) });
   381→    venc.encode(vf, { keyFrame: i % (fps * 2) === 0 }); vf.close();
   382→    if (!noSeek && i + 1 < n) seekP = seekTo(seg.start + (i + 1) * spd / fps, v);
   383→    while (venc.encodeQueueSize > 14) await sleep(2);
   384→    slot.fr = i + 1;
   385→    doneCb && doneCb();
   386→  }
   387→  return n;
   388→}

   390→/* ---------- WORKER SATU PART (v2.2 TURBO) ----------
   391→   Elemen video sendiri + canvas sendiri + muxer sendiri.
   392→   Frame diambil sambil video diputar (tanpa seek per frame). */
   393→async function renderPartWorker(seg, srcUrl, W, H, fps, br, vCfg, aCodec, slot, doneCb, mergeCtx = null) {
   394→  /* v2.12: KECEPATAN video — 1 = normal, 2 = 2× cepat (part ½ lebih pendek),
   395→     0.5 = slow-motion. Durasi output = durasi sumber ÷ spd. */
   396→  const spd = clamp(state.speed || 1, 0.25, 4);
   397→  const durS = (seg.end - seg.start) / spd;
   398→  const v = document.createElement('video');
   399→  v.muted = true; v.playsInline = true; v.preload = 'auto';
   400→  v.style.cssText = 'position:fixed;left:-9999px;top:-9999px;width:2px;height:2px;opacity:0';
   401→  v.src = srcUrl;
   402→  document.body.appendChild(v);
   403→  v.addEventListener('error', () => console.error(`part ${seg.i + 1} media error: code=${v.error && v.error.code} msg=${v.error && v.error.message}`));
   404→  let venc = null, _stage = 'init', n = 0;
   405→  let audioP = null, audioErr = null;   /* v3.0: audio paralel dgn video */
   406→  try {
   407→    await new Promise((res, rej) => {
   408→      const ok = () => { clean(); res(); };
   409→      const bad = () => { clean(); rej(new Error('worker: meta tidak termuat')); };
   410→      const clean = () => { v.removeEventListener('loadedmetadata', ok); v.removeEventListener('error', bad); };
   411→      v.addEventListener('loadedmetadata', ok); v.addEventListener('error', bad);
   412→      setTimeout(() => { if (v.readyState >= 1) ok(); }, 6000);
   413→    });

   415→    await seekTo(seg.start, v);
   416→
   417→    _stage = 'canvas';
   418→    const ecv = document.createElement('canvas'); ecv.width = W; ecv.height = H;
   419→    const ec = ecv.getContext('2d', { alpha: false });
   420→    _stage = 'muxer';
   421→    /* v2.15: part BESAR ditulis LANGSUNG ke disk (StreamTarget → IPC 8 MB) —
   422→       dulu SELURUH berkas MP4 menumpuk di RAM (ArrayBufferTarget) sampai
   423→       render selesai; part 30-60 mnt = gigabyte per part di memori,
   424→       batch film panjang lalu kehabisan memori → ekspor merambat.
   425→       Part kecil tetap in-memory (fastStart penuh, kompatibilitas maks).
   426→       v3.0: GABUNG VIDEO — semua part memakai SATU muxer bersama milik
   427→       driver merge (satu file MP4 kontinu di disk); worker tidak menulis
   428→       file sendiri, tidak finalize — hanya meng-encode. */
   429→    const nm = partFileName(state.title, state.partPrefix, seg.i + 1);
   430→    const estBytes = Math.round(durS * (br + (aCodec ? 160000 : 0)) / 8) + 4e6;
   431→    const streamDisk = !mergeCtx && estBytes > (state._streamThreshMB || 384) * 1048576;
   432→    let wsId = null, wsErr = null, wsPath = mergeCtx ? mergeCtx.outPath : nm, chain = Promise.resolve(), sent = 0;
   433→    const mkAudio = aCodec ? { audio: { codec: aCodec === 'mp4a.40.2' ? 'aac' : 'opus', numberOfChannels: 2, sampleRate: 48000 } } : {};
   434→    const mkVideo = { codec: 'avc', width: W, height: H, frameRate: fps };
   435→    let muxer;
   436→    if (mergeCtx) {
   437→      muxer = mergeCtx.muxer;
   438→      wsErr = mergeCtx.wsErr;
   439→    } else if (streamDisk) {
   440→      const w = await window.kinostra.beginWrite(slot.dir, nm);
   441→      wsId = w.id; wsPath = w.path;
   442→      console.info(`part ${seg.i + 1}: muxer streaming disk aktif (±${fmtMB(estBytes)}) — RAM tetap datar (v2.15)`);
   443→      muxer = new Mp4Muxer.Muxer({
   444→        target: new Mp4Muxer.StreamTarget({
   445→          chunked: true, chunkSize: 8 * 1024 * 1024,
   446→          /* onData(data, posisi) — tulis-acak ke disk via pwrite (IPC);
   447→             rantai `chain` menjaga urutan panggilan, patch header mdat di
   448→             posisi lama oleh mp4-muxer (fastStart:false) tetap benar */
   449→          onData: (data, pos) => {
   450→            if (wsErr) return;
   451→            sent += data.byteLength;
   452→            slot.saveProg = Math.min(0.98, sent / estBytes);
   453→            chain = chain.then(() => window.kinostra.writeChunkAt(wsId, data, pos)).catch(e => { wsErr = wsErr || e; });
   454→          }
   455→        }),
   456→        fastStart: false, firstTimestampBehavior: 'offset',
   457→        video: mkVideo, ...mkAudio
   458→      });
   459→    } else {
   460→      muxer = new Mp4Muxer.Muxer({
   461→        target: new Mp4Muxer.ArrayBufferTarget(), fastStart: 'in-memory', firstTimestampBehavior: 'offset',
   462→        video: mkVideo, ...mkAudio
   463→      });
   464→    }
   465→    if (aCodec) {
   466→      _stage = 'audio';
   467→      /* v2.15: encode audio PER BLOK — RAM datar di part panjang
   468→         v3.0 OPTIMASI KECEPATAN: audio di-encode BERSAMAAN dengan render
   469→         video (dulu audio selesai dulu → baru video: waktu menjumlah).
   470→         Kedua encoder mengalir ke muxer yang sama — mp4-muxer menerima
   471→         potongan video/audio ter-selang; hasil identik, ±30-45% lebih cepat. */
   472→      audioP = withTimeout(
   473→        encodeAudioStream(muxer, seg, durS, spd, aCodec, mergeCtx ? mergeCtx.off : 0),
   474→        300000, 'encoding audio macet'
   475→      ).catch(e => { audioErr = audioErr || e; });
   476→    }
   477→    _stage = 'configure';
   478→    const hw = vCfg.hardwareAcceleration === 'prefer-hardware';
   479→    /* v2.7 FIX "SUARA JALAN, GAMBAR BEKU": encoder GPU Windows (Media
   480→       Foundation) dengan latencyMode 'quality' dapat mengeluarkan chunk
   481→       B-frame berurutan DECODE (timestamp bolak-balik). mp4-muxer melempar
   482→       error monotonic DI DALAM callback output → pengecualian tertelan
   483→       diam-diam → hanya beberapa frame pertama masuk MP4 → gambar beku +
   484→       audio normal + ekspor "berhasil". Perisai 4 lapis:
   485→       1) latencyMode 'realtime' utk SEMUA encoder (tanpa reorder B-frame)
   486→       2) timestamp turun > 5 ms terdeteksi → gagal cepat → auto-retry SW
   487→       3) error encoder diteruskan ke promise (bukan hanya console)
   488→       4) verifikasi jumlah chunk video setelah flush (≥ 90% frame) */
   489→    let vChunks = 0, lastVts = -Infinity, vencErrMsg = null;
   490→    venc = new VideoEncoder({
   491→      output: (c, m) => {
   492→        const ts = typeof c.timestamp === 'number' ? c.timestamp : 0;
   493→        if (vChunks > 0 && ts <= lastVts) {
   494→          if (lastVts - ts > 5000) {
   495→            vencErrMsg = `encoder mengeluarkan frame dengan urutan waktu tidak naik (${Math.round(lastVts / 1000)} → ${Math.round(ts / 1000)} ms)`;
   496→            console.error('venc-stream', vencErrMsg);
   497→          }
   498→          return; /* jangan masukkan chunk rusak/nol-durasi ke muxer */
   499→        }
   500→        lastVts = ts; vChunks++;
   501→        muxer.addVideoChunk(c, m);
   502→      },
   503→      error: e => { vencErrMsg = (e && e.message) || String(e); console.error('venc', vencErrMsg); }
   504→    });
   505→    venc.configure({ ...vCfg, latencyMode: 'realtime' });
   506→    const CAP = hw ? 40 : 14;
   507→
   508→    /* gambar 1 frame pada waktu media m → encode dengan timestamp nyata
   509→       v2.12: PTS = waktu OUTPUT = (m − mulai part) ÷ kecepatan
   510→       v3.0: + offset global mergeCtx.off saat GABUNG VIDEO */
   511→    const drawAt = (t, ts) => {
   512→      setCompSrc(v);
   513→      drawComposition(ec, W, H, t);
   514→      setCompSrc(null);
   515→      const vf = new VideoFrame(ecv, { timestamp: ts, duration: Math.round(1e6 / fps) });
   516→      venc.encode(vf, { keyFrame: n % (fps * 2) === 0 }); vf.close();
   517→      n++; slot.fr = n;
   518→    };
   519→    const outTs = (m) => Math.max(0, Math.round((((mergeCtx ? mergeCtx.off : 0) + (m - seg.start) / spd) * 1e6)));
   520→
   521→    if (RVFC_OK && !state.isAudio) {
   522→      /* ============ v2.2: CAPTURE PLAYBACK (tanpa seek per frame) ============
   523→         v2.5: + WATCHDOG ANTI-MACET. Penyebab lama ekspor "macet di bagian
   524→         akhir tanpa pesan": frame callback (rVFC) berhenti memicu di frame
   525→         terakhir & event ended tidak sampai, atau encoder GPU berhenti
   526→         mengeluarkan output → promise render tidak pernah selesai.
   527→         Sekarang interval 250 ms memantau: video selesai? waktu media
   528→         sudah mencapai akhir part? video error? encoder diam terlalu
   529→         lama? → semua diarahkan SELESAI atau GAGAL DENGAN PESAN JELAS. */
   530→      _stage = 'stream';
   531→      /* v2.11 FIX "KADANG GERAK KADANG BEKU": dulu capRate mulai 2× dan bisa
   532→         naik sampai 4× — di atas kemampuan decoder/compositor, frame sumber
   533→         dibuang beramai-ramai, mediaTime antar frame tersaji LOMPAT, PTS
   534→         ekspor ikut bolong → pemutar menahan frame lama berdetik-detik.
   535→         Sekarang: mulai 1.5×, turun SEGERA saat ada frame hilang, dan setiap
   536→         lompatan DIISI ULANG dengan seek presisi (refillGap) sehingga aliran
   537→         PTS selalu rapat & mulus.
   538→         v2.14: bila encoder GPU (hardware) aktif, langit-langit dinaikkan
   539→         ke 3× — encoding bukan lagi hambatan, decoder-lah penentu. */
   540→      const RATE_MAX = hw ? 3 : (_xPar >= 3 ? 2 : 2.5);
   541→      /* v2.12: toleransi & target lompatan diskalakan kecepatan —
   542→         pada spd 2× jarak sumber antar frame tampil 2× lebih renggang */
   543→      const frameGapTol = 1.8 / fps * spd;
   544→      let capRate = 1.5, dAcc = 0, dN = 0, lastM = -1, hadFirst = false;
   545→      const applyRate = () => { try { v.playbackRate = Math.min(4.2, capRate * spd); } catch (e) { } };
   546→      applyRate();
   547→      let lastAct = Date.now(), qStallSince = 0;
   548→      let refilling = false, refillN = 0, maxGapMs = 0;
   549→      /* v2.11: isi ulang frame yang hilang di interval (from, to) — seek
   550→         presisi slot per slot, encode dengan PTS aslinya. Dipanggil SEBELUM
   551→         frame@to di-encode agar urutan PTS tetap naik. lastAct disegarkan
   552→         tiap slot supaya watchdog 30 dtk tidak mengira render diam. */
   553→      const refillGap = async (from, to) => {
   554→        const step = spd / fps;
   555→        let t = from + step, k = 0;
   556→        while (t < to - step * 0.5) {
   557→          if (state.abort) throw new Error('Dibatalkan');
   558→          if (state.exportFatal) throw state.exportFatal;
   559→          await seekTo(Math.min(t, Math.max(0, (v.duration || seg.end) - 0.011)), v);
   560→          drawAt(t, outTs(t));
   561→          lastAct = Date.now(); k++; t += step;
   562→          doneCb && doneCb();
   563→        }
   564→        refillN += k; return k;
   565→      };
   566→      await new Promise((resolve, rej) => {
   567→        let finished = false;
   568→        const stop = () => { try { v.pause(); } catch (e) { } };
   569→        const fin = () => { if (finished) return; finished = true; stop(); clearInterval(watch); resolve(); };
   570→        const fail = (err) => { if (finished) return; finished = true; stop(); clearInterval(watch); rej(err); };
   571→        /* --- v2.5: WATCHDOG (detak 250 ms) --- */
   572→        const watch = setInterval(() => {
   573→          if (finished) { clearInterval(watch); return; }
   574→          if (state.abort) return fail(new Error('Dibatalkan'));
   575→          if (state.exportFatal) return fail(state.exportFatal);   /* v2.10: part lain sudah gagal */
   576→          /* v2.7: error encoder terdeteksi → gagal cepat → retry software */
   577→          if (vencErrMsg) return fail(new Error('encoder video bermasalah: ' + vencErrMsg));
   578→          if (v.error) return fail(new Error('video rusak/tak terbaca: ' + (v.error.message || ('kode ' + v.error.code))));
   579→          /* KUNCI FIX: tangani akhir video walau frame callback mati —
   580→             inilah yang dulu bikin macet selamanya di bagian akhir */
   581→          if (v.ended || v.currentTime >= seg.end - 0.004) return fin();
   582→          const now = Date.now();
   583→          if (venc.encodeQueueSize > CAP) {
   584→            if (!qStallSince) qStallSince = now;
   585→            else if (now - qStallSince > 45000)
   586→              return fail(new Error('encoder video macet (antrean tidak mengalir 45 dtk) — akan dicoba ulang otomatis dengan encoder software'));
   587→          } else qStallSince = 0;
   588→          if (now - lastAct > 30000)
   589→            return fail(new Error('render diam total 30 dtk — akan dicoba ulang otomatis dengan encoder software'));
   590→          /* dorong ulang bila video diam padahal encoder tidak penuh */
   591→          if (v.paused && !refilling && !v.seeking && venc.encodeQueueSize <= CAP && !v.ended) {
   592→            try { const p = v.play(); if (p && p.catch) p.catch(err => fail(new Error('video tidak bisa dilanjut: ' + (err.message || err)))); }
   593→            catch (err) { fail(new Error('video tidak bisa dilanjut: ' + (err.message || err))); }
   594→          }
   595→        }, 250);
   596→        const onFrame = async (now, meta) => {
   597→          try {
   598→            if (finished) return;
   599→            lastAct = Date.now();
   600→            if (state.abort) return fail(new Error('Dibatalkan'));
   601→            if (state.exportFatal) return fail(state.exportFatal);   /* v2.10 */
   602→            const m = meta.mediaTime;
   603→            /* v3.1 ANTI-GLITCH: laporan mediaTime yang TIDAK WAJAR (melompat
   604→               jauh melebihi posisi putar elemen — decoder/kfile sesekali
   605→               melaporkan durasi file sebagai mediaTime) TIDAK boleh dianggap
   606→               akhir part & tidak di-encode. Dulu ini bikin ekspor "selesai"
   607→               berisi cuma 3-4 frame (video+audio terpotong serentak). */
   608→            if (hadFirst && !v.ended && !v.seeking && m - lastM > 1.0 && Math.abs(m - v.currentTime) > 0.75) {
   609→              return pump();   /* buang laporan palsu, minta frame berikutnya */
   610→            }
   611→            if (v.ended || m >= seg.end - 0.004) return fin();
   612→            if (m >= seg.start - 0.06 && m > lastM) {
   613→              /* v2.11: DETEKSI FRAME HILANG — lompatan mediaTime > 1.8× durasi
   614→                 frame berarti decoder/compositor membuang frame. Isi slot yang
   615→                 bolong lewat seek presisi SEBELUM frame m di-encode supaya
   616→                 urutan PTS rapat (tanpa durasi sampel multi-detik = beku). */
   617→              const gap = hadFirst ? m - lastM : 0;
   618→              if (gap > frameGapTol) {
   619→                maxGapMs = Math.max(maxGapMs, gap * 1000);
   620→                refilling = true; stop();
   621→                const k = await refillGap(lastM, m);
   622→                if (finished) return;
   623→                await seekTo(Math.min(m, Math.max(0, (v.duration || seg.end) - 0.011)), v);
   624→                refilling = false;
   625→                if (finished) return;
   626→                /* decoder kewalahan pada rate ini → turun SEGERA */
   627→                if (capRate > 1) { capRate = Math.max(1, capRate - 0.5); applyRate(); }
   628→                dAcc = 0; dN = 0;
   629→              }
   630→              drawAt(m, outTs(m));
   631→              hadFirst = true;
   632→              /* rate adaptif: naik HANYA bila jendela bersih tanpa lompatan
   633→                 v2.12: target jarak antar frame = spd/fps (waktu sumber) */
   634→              if (lastM >= 0) {
   635→                const d = m - lastM;
   636→                if (d > 0 && d < 0.6 * spd) {
   637→                  dAcc += d; dN++;
   638→                  if (dN >= 24) {
   639→                    const avg = dAcc / dN; dAcc = 0; dN = 0; const tg = spd / fps;
   640→                    if (avg > tg * 1.5 && capRate > 1) { capRate = Math.max(1, capRate - 0.5); applyRate(); }
   641→                    else if (avg < tg * 1.18 && capRate < RATE_MAX) { capRate = Math.min(RATE_MAX, capRate + 0.5); applyRate(); }
   642→                  }
   643→                }
   644→              }
   645→              lastM = m;
   646→              doneCb && doneCb();
   647→            }
   648→            /* backpressure: pause bila encoder tertinggal, lanjut otomatis */
   649→            if (venc.encodeQueueSize > CAP) { stop(); setTimeout(pump, 5); return; }
   650→            pump();
   651→          } catch (e) {
   652→            refilling = false;
   653→            fail(e && e.message ? e : new Error(String(e)));
   654→          }
   655→        };
   656→        const pump = () => {
   657→          if (finished) return;
   658→          if (state.abort) return fail(new Error('Dibatalkan'));
   659→          if (state.exportFatal) return fail(state.exportFatal);   /* v2.10 */
   660→          if (v.ended || lastM >= seg.end - 0.004) return fin();
   661→          if (v.paused) {
   662→            try { const p = v.play(); if (p && p.catch) p.catch(err => fail(new Error('video tidak bisa diputar: ' + (err.message || err)))); }
   663→            catch (err) { return fail(new Error('video tidak bisa diputar: ' + (err.message || err))); }
   664→          }
   665→          try { v.requestVideoFrameCallback(onFrame); } catch (e) { fin(); }
   666→        };
   667→        v.addEventListener('ended', fin, { once: true });
   668→        v.addEventListener('error', () => fail(new Error('video error saat dirender')), { once: true });
   669→        pump();
   670→      });
   671→      if (refillN) console.info(`part ${seg.i + 1}: ${n} frame dirender · ${refillN} diisi-ulang (lompatan maks ${Math.round(maxGapMs)} ms) — anti-beku v2.11 aktif`);
   672→      if (n === 0) {
   673→        /* sumber aneh / tidak menghasilkan frame — jalankan jalur lama */
   674→        _stage = 'fallback-seek';
   675→        await renderFramesBySeek(v, ecv, ec, venc, seg, fps, slot, doneCb, false, spd, mergeCtx ? mergeCtx.off : 0);
   676→        n = slot.fr;
   677→      }
   678→    } else {
   679→      /* audio-only / tanpa rVFC: jalur frame loop (tanpa seek utk audio) */
   680→      _stage = 'frames';
   681→      await renderFramesBySeek(v, ecv, ec, venc, seg, fps, slot, doneCb, !!state.isAudio, spd, mergeCtx ? mergeCtx.off : 0);
   682→      n = slot.fr;
   683→    }
   684→
   685→    _stage = 'flush';
   686→    slot.stage = 'flush';
   687→    /* v3.0: tunggu encoder audio paralel selesai sebelum flush video */
   688→    if (audioP) { await audioP; if (audioErr) throw audioErr; }
   689→    /* v2.5: flush dengan batas waktu — encoder yang hang tidak lagi
   690→       membuat ekspor macet selamanya tanpa pesan */
   691→    await withTimeout(venc.flush(), 60000, 'encoder video tidak selesai (flush macet 60 dtk)');
   692→    venc.close(); venc = null;
   693→    /* v2.7: VERIFIKASI ALIRAN VIDEO — tangkap dua mode kegagalan encoder
   694→       GPU yang "diam": (a) error/reorder terdeteksi saat merender,
   695→       (b) output kurang dari 90% frame yang dirender. Tanpa ini MP4 tetap
   696→       tersimpan tapi isinya gambar beku. Gagal di sini → auto-retry dengan
   697→       encoder software oleh exportPartsToDir. */
   698→    if (vencErrMsg) throw new Error('aliran video tidak valid: ' + vencErrMsg);
   699→    /* v2.11: verifikasi diperketat 90% → 98% — encoder yang membuang output
   700→       di tengah stream kini terdeteksi → auto-retry software, bukan MP4 beku */
   701→    const minChunks = Math.max(1, Math.ceil(n * 0.98));
   702→    if (vChunks < minChunks)
   703→      throw new Error(`encoder hanya mengeluarkan ${vChunks} dari ${n} frame (gambar akan beku)`);
   704→    _stage = 'finalize';
   705→    slot.stage = 'finalize';
   706→    if (wsErr) throw wsErr;
   707→    if (mergeCtx && mergeCtx.wsErr) throw mergeCtx.wsErr;
   708→    if (mergeCtx) {
   709→      /* v3.0: part gabung — driver merge yang mem-finalize & menyimpan */
   710→      slot.prog = 1; slot.fr = slot.total = Math.max(slot.total, n);
   711→      slot.saveProg = 1;
   712→      return { name: mergeCtx.outName, path: mergeCtx.outPath, size: 0, merged: true };
   713→    }
   714→    await withTimeout(Promise.resolve().then(() => muxer.finalize()), 30000, 'penulisan header MP4 macet');
   715→    _stage = 'save';
   716→    slot.stage = 'save';
   717→    let wr;
   718→    if (streamDisk) {
   719→      /* v2.15: semua byte sudah mengalir ke disk SELAMA render — tinggal
   720→         menuntaskan rantai tulis, menutup stream, membaca ukuran berkas */
   721→      if (wsErr) throw wsErr;
   722→      await withTimeout(chain, 180000, 'menulis file ke disk macet (cek ruang kosong disk)');
   723→      if (wsId) { const id = wsId; wsId = null; await window.kinostra.endWrite(id); }
   724→      const stt = await window.kinostra.stat(wsPath);
   725→      wr = { path: wsPath, size: (stt && stt.ok) ? stt.size : sent };
   726→    } else {
   727→      wr = await withTimeout(
   728→        saveBufferToDir(muxer.target.buffer, slot.dir, nm, (w, t) => { slot.saveProg = w / t; }),
   729→        180000, 'menyimpan file ke disk macet (cek ruang kosong disk)');
   730→    }
   731→    slot.prog = 1; slot.fr = slot.total = Math.max(slot.total, n);
   732→    return { name: wr.path.split(/[\\/]/).pop(), path: wr.path, size: wr.size };
   733→  } catch (err) {
   734→    if (_stage !== 'frames' && _stage !== 'stream') console.error(`part ${seg.i + 1} GAGAL di tahap ${_stage}: ${err && err.name || 'ERR'} | ${err && err.message || err}`);
   735→    throw err;
   736→  } finally {
   737→    try { if (venc && venc.state !== 'closed') venc.close(); } catch (e) { }
   738→    /* v2.15: bila gagal di tengah render streaming → buang berkas setengah jadi */
   739→    try { if (wsId) { const id = wsId; wsId = null; window.kinostra.abortWrite(id).catch(() => { }); } } catch (e) { }
   740→    try { v.pause(); v.removeAttribute('src'); v.load(); } catch (e) { }
   741→    try { v.remove(); } catch (e) { }
   742→  }
   743→}

   745→/* ---------- INTI RENDER: SEMUA PART PARALEL -> folder dir ----------

   747→async function exportPartsToDir(dir, ui) {

   750→     loading) sudah selesai sebelum menentukan ada tidaknya track audio */

   752→  /* v3.1: ZOOM PEMBICARA — analisis otomatis bila setelan aktif & video
   753→     belum dianalisis (berlaku untuk ekspor tunggal, split & BATCH EKSPOR:
   754→     semua video di kotak otomatis dianalisis satu per satu saat diproses).
   755→     Gagal internal (tanpa internet pertama kali dsb.) → lanjut TANPA zoom. */
   756→  if (state.spk && state.spk.on && !state.isAudio && !state.spk.ready && !state.spk.scanning) {

   781→     suara asli + backsound tetap diekspor DENGAN audio) */

   790→

   793→  const totalFr = slots.reduce((a, s) => a + s.total, 0);
   794→  const t0 = performance.now();

   799→    lastUi = now;

   801→    if (ui.prog) ui.prog(done / totalFr);
   802→    if (ui.sub) {
   803→      const el = Math.max(0.001, (now - t0) / 1000);
   804→      const rt = (done / fps / el);
   805→      const parts = slots.map(s => {
   806→        const pct = Math.round(100 * Math.min(s.fr, s.total) / s.total);
   807→        const stg = s.stage === 'flush' ? '· FLUSH' : s.stage === 'finalize' ? '· FINAL' : s.stage === 'save' ? '· SIMPAN' : '';
   808→        const tag = s.retried && s.working ? ' (ULANGI-SW)' : '';
   809→        return s.working ? `P${String(s.seg.i + 1).padStart(2, '0')} ${pct}%${tag}${stg}` : null;
   810→      }).filter(Boolean).join(' · ');
   811→      ui.sub(`${parts || 'menunggu…'}${rt > 0.05 ? ` · ${rt.toFixed(1)}× realtime` : ''}`);
   812→    }
   813→  };
   814→
   815→  /* v2.6: file virtual (impor/batch streaming) — worker memutar langsung
   816→     dari disk via kfile:// (Range), tanpa menyalin file ke memori
   817→     v3.0: dibuat LAZY — LANJUTKAN EKSPOR bisa melewati semua part tanpa
   818→     menyentuh src (state.file uji/korup tidak membuat createObjectURL meledak) */
   819→  let _srcUrl = null;
   820→  const getSrcUrl = () => {
   821→    if (_srcUrl == null) _srcUrl = state.file.virtual ? state.file.kurl : URL.createObjectURL(state.file);
   822→    return _srcUrl;
   823→  };
   824→  const results = new Array(segs.length).fill(null);
   825→  state.exportFatal = null;   /* v2.10: mulai ekspor baru — bersihkan sisa gagal lama */
   826→  try {
   827→    if (ui.title) ui.title(segs);
   828→    if (ui.prog) ui.prog(0);
   829→    if (ui.sub) ui.sub(`encoder ${enc.kind === 'hw' ? `GPU ${enc.fps}fps` : `CPU ${enc.fps}fps`} · ${par} render paralel${RVFC_OK && !state.isAudio ? ' · turbo stream' : ''}${spd !== 1 ? ` · kecepatan ${spd}×` : ''}…`);
   830→
   831→    /* jalankan worker dengan batas `par` sekaligus (antrean)
   832→       v2.10: kegagalan part TIDAK lagi men-set state.abort (flag batal
   833→       milik pengguna) — cukup state.exportFatal supaya worker lain berhenti.
   834→       Penelepon (ekspor tunggal / batch) menerima Error aslinya dan bisa
   835→       menampilkan penyebabnya; batch menandai video GAGAL lalu LANJUT. */
   836→    let cursor = 0;
   837→    const launch = async () => {
   838→      while (cursor < slots.length && !xAborted()) {
   839→        const idx = cursor++;
   840→        const slot = slots[idx];
   841→        /* v3.0: LANJUTKAN EKSPOR — lewati part yang filenya SUDAH ADA di
   842→           folder output (hasil ekspor sebelumnya yang tertidur/gagal di
   843→           tengah). File part utuh = tanda part itu selesai dirender. */
   844→        if (state.resumeExport && !state.isAudio) {

   847→            const st0 = await window.kinostra.statJoin(dir, nm0);
   848→            if (st0 && st0.ok && st0.size > 51200) {

   854→          } catch (e) { }   /* statJoin tak tersedia / gagal → render normal */
   855→        }
   856→        /* v3.0: KOMPRES PINTAR — bitrate dihitung dari TARGET UKURAN per part
   857→           (bitrate = byte target ÷ durasi output − bitrate audio) */
   858→        let cfgPart = vCfg, swPart = swCfg;
   859→        if (state.targetMB > 0) {
   860→          const durP = Math.max(0.5, (slot.seg.end - slot.seg.start) / spd);
   861→          const brP = Math.round(clamp((state.targetMB * 8388608 * 0.93) / durP - 160000, 150000, 20e6));
   862→          cfgPart = { ...vCfg, bitrate: brP };

   864→        }
   865→        slot.working = true; uiTick(true);
   866→        try {
   867→          results[idx] = await renderPartWorker(slot.seg, getSrcUrl(), W, H, fps, br, cfgPart, aCodec, slot, () => uiTick(false));
   868→        } catch (e) {
   869→          /* v2.5: RETRY OTOMATIS 1× dengan encoder SOFTWARE — penyembuh
   870→             utama "ekspor macet di bagian akhir" (encoder GPU berhenti).
   871→             Jangan ulangi bila user membatalkan sendiri / part lain gagal.
   872→             v2.15: kegagalan jalur hardware dicatat — 2× jatuh → sisa batch
   873→             otomatis memakai software (anti-pelan, tanpa dobel kerja). */
   874→          if (hw && !xAborted()) noteHwFailure(e && e.message || String(e));
   875→          if (!xAborted() && swCfg) {
   876→            console.warn(`part ${idx + 1} gagal (${e && e.message || e}) — ulangi 1× dengan encoder software`);
   877→            slot.fr = 0; slot.stage = null; slot.saveProg = 0; slot.retried = true; slot.working = true; uiTick(true);
   878→            try {
   879→              results[idx] = await renderPartWorker(slot.seg, getSrcUrl(), W, H, fps, br, swPart || swCfg, aCodec, slot, () => uiTick(false));
   880→              console.info(`part ${idx + 1} sukses pada percobaan ke-2 (encoder software)`);
   881→            } catch (e2) {
   882→              const det2 = e2 && (e2.name + ' | ' + e2.message) || String(e2);
   883→              console.error(`part ${idx + 1} GAGAL juga di percobaan ke-2: ${det2}`);
   884→              if (!xAborted()) state.exportFatal = e2;
   885→              slot.working = false; uiTick(true);
   886→              return;
   887→            }
   888→          } else {
   889→            const det = e && (e.name + ' | ' + e.message + ' | ' + String(e.stack || '').split('\n')[1] || '') || String(e);
   890→            console.error(`part ${idx + 1} GAGAL: ${det}`);
   891→            if (!xAborted()) state.exportFatal = e;
   892→            slot.working = false; uiTick(true);
   893→            return;
   894→          }
   895→        }
   896→        slot.working = false; uiTick(true);
   897→      }
   898→    };
   899→    const runners = [];
   900→    for (let k = 0; k < par; k++) runners.push(launch());
   901→    /* pantau progres selagi worker jalan
   902→       v2.5: + WATCHDOG GLOBAL — bila TIDAK ADA kemajuan apa pun dari semua
   903→       worker selama 45 dtk, hentikan dengan pesan jelas (bukan macam) */
   904→    await new Promise(res => {
   905→      let lastSig = '', lastSigAt = Date.now();
   906→      const iv = setInterval(() => {
   907→        uiTick(false);
   908→        if (xAborted() || results.every(r => r)) { clearInterval(iv); res(); return; }
   909→        const sig = slots.map(s => `${s.fr}:${s.working ? 1 : 0}:${s.stage || ''}:${Math.round((s.saveProg || 0) * 100)}`).join('|');
   910→        const now = Date.now();
   911→        if (sig !== lastSig) { lastSig = sig; lastSigAt = now; }
   912→        else if (now - lastSigAt > 45000) {
   913→          clearInterval(iv);
   914→          /* v2.10: watchdog = kegagalan internal, BUKAN batal pengguna */
   915→          if (!xAborted()) {
   916→            state.exportFatal = new Error('Render berhenti merespons 45 detik — dihentikan otomatis. Coba set PARALEL 1× atau kualitas lebih rendah lalu ulangi.');
   917→          }
   918→          res();
   919→        }
   920→      }, 150);
   921→    });
   922→    await Promise.all(runners).catch(() => { });
   923→    if (state.exportFatal) { const err = state.exportFatal; throw err; }
   924→    if (state.abort && !results.every(r => r)) {
   925→      const made = results.filter(Boolean);
   926→      throw new Error(made.length ? `Dibatalkan — ${made.length} part tersimpan` : 'Dibatalkan');
   927→    }
   928→    if (results.some(r => !r)) throw new Error('Sebagian part gagal dirender');
   929→    if (ui.sub) {
   930→      const el = (performance.now() - t0) / 1000;
   931→      ui.sub(`selesai dalam ${el < 90 ? el.toFixed(0) + ' detik' : (el / 60).toFixed(1) + ' menit'} · ${par}× paralel${hw ? ' · GPU' : ''}`);
   932→    }
   933→    if (ui.prog) ui.prog(1);
   934→    return results;
   935→  } finally {
   936→    state.exportFatal = null;   /* v2.10: jangan bocor ke ekspor/video berikutnya */
   937→    if (_srcUrl != null && !state.file.virtual) try { URL.revokeObjectURL(_srcUrl); } catch (e) { }
   938→  }
   939→}

  1045→  await ensureFontsReady();
  1046→  if (!paths || paths.length < 2) throw new Error('Pilih minimal 2 video di Kotak Video untuk digabung');
  1047→  const fps = state.fps, sc = parseFloat(state.scale), q = QUAL[state.quality];
  1048→  const spd = clamp(state.speed || 1, 0.25, 4);
  1049→  const dims = outDims(sc), { W, H } = dims;
  1050→  const br = Math.round(q.br * (sc === 1 ? 1 : sc === 0.75 ? 0.62 : 0.38));
  1051→  const vCfg = await pickVCodecCfg(W, H, br, fps);
  1052→  if (!vCfg) throw new Error('Tidak ada codec video yang didukung engine');
  1053→  const swCfg = await pickSWCfg(W, H, br, fps);
  1054→  const aCodec = await pickACodec();
  1055→  if (!aCodec) throw new Error('Encoder audio tidak tersedia');
  1056→  const nm = `GABUNGAN_${paths.length}_VIDEO_${slug(state.batch.find(b => b.path === paths[0])?.name || 'FILM').slice(0, 24)}.mp4`;
  1057→  /* mulai file gabungan (streaming disk — RAM datar) */
  1058→  const w = await window.kinostra.beginWrite(dir, nm);
  1059→  const mergeCtx = { muxer: null, off: 0, outName: nm, outPath: w.path, wsId: w.id, chain: Promise.resolve(), sent: 0, wsErr: null };
  1060→  mergeCtx.muxer = new Mp4Muxer.Muxer({
  1061→    target: new Mp4Muxer.StreamTarget({
  1062→      chunked: true, chunkSize: 8 * 1024 * 1024,
  1063→      onData: (data, pos) => {
  1064→        if (mergeCtx.wsErr) return;
  1065→        mergeCtx.sent += data.byteLength;
  1066→        mergeCtx.chain = mergeCtx.chain.then(() => window.kinostra.writeChunkAt(mergeCtx.wsId, data, pos)).catch(e => { mergeCtx.wsErr = mergeCtx.wsErr || e; });
  1067→      }
  1068→    }),
  1069→    fastStart: false, firstTimestampBehavior: 'offset',
  1070→    video: { codec: 'avc', width: W, height: H, frameRate: fps },
  1071→    audio: { codec: aCodec === 'mp4a.40.2' ? 'aac' : 'opus', numberOfChannels: 2, sampleRate: 48000 }
  1072→  });
  1073→  state._noPart = true;          /* label PART tidak masuk file gabungan */
  1074→  const prevFatal = state.exportFatal;
  1075→  state.exportFatal = null;
  1076→  const t0 = performance.now();
  1077→  let doneParts = 0;
  1078→  try {
  1079→    if (ui.prog) ui.prog(0);
  1080→    if (ui.sub) ui.sub(`Menyiapkan encoder ${_lastEnc.kind === 'hw' ? 'GPU' : 'CPU'} · ${paths.length} video…`);
  1081→    for (let i = 0; i < paths.length; i++) {
  1082→      if (state.abort) throw new Error('Dibatalkan');
  1083→      if (state.exportFatal) throw state.exportFatal;
  1084→      if (mergeCtx.wsErr) throw mergeCtx.wsErr;
  1085→      const nmv = paths[i].split(/[\\/]/).pop();
  1086→      if (ui.sub) ui.sub(`[${i + 1}/${paths.length}] Memuat ${nmv}…`);
  1087→      /* lepaskan buffer film sebelumnya (hemat RAM, sama dgn batch) */
  1088→      state.audioBuffer = null; state.peaks = null; state.audioRms = 0;
  1089→      await loadFromPath(paths[i], true);
  1090→      if (!state.file || state.file.path !== paths[i]) throw new Error(`video ${i + 1} (${nmv}) tidak bisa dimuat`);
  1091→      if (!state.duration) throw new Error(`durasi ${nmv} tidak terbaca — format tak didukung`);
  1092→      if (ui.sub) ui.sub(`[${i + 1}/${paths.length}] ${nmv} · ${fmtT(state.duration)} · memeriksa subtitel…`);
  1093→      await ensureAutoSubs({ sub: s => { if (ui.sub) ui.sub(`[${i + 1}/${paths.length}] ${s}`); }, prog: () => { } });
  1094→      /* v3.1: ZOOM PEMBICARA per film — tiap video di kotak dianalisis
  1095→         sendiri (pembicara tiap film berbeda) bila setelan aktif */
  1096→      if (state.spk && state.spk.on && !state.isAudio && !state.spk.ready && state.spk.auto && !state.spk.scanning) {
  1097→        try {
  1098→          if (ui.sub) ui.sub(`[${i + 1}/${paths.length}] ${nmv} · menganalisis pembicara…`);
  1099→          await analyzeSpeakers({
  1100→            prog: p => { if (ui.prog) ui.prog(clamp(((i + p) / paths.length) * 0.5, 0, 1)); },
  1101→            stat: s => { if (ui.sub) ui.sub(`[${i + 1}/${paths.length}] ${s}`); }
  1102→          });
  1103→        } catch (e) { console.warn('analisis pembicara gagal utk', nmv, e && e.message); }
  1104→      }
  1105→      if (state.music.buffer) await composeMusic(false, true, true);
  1106→      const srcUrl = state.file.virtual ? state.file.kurl : URL.createObjectURL(state.file);
  1107→      try {
  1108→        const segs = segments();
  1109→        for (const seg of segs) {
  1110→          if (state.abort) throw new Error('Dibatalkan');
  1111→          if (state.exportFatal) throw state.exportFatal;
  1112→          const slot = { seg, dir, prog: 0, fr: 0, total: Math.max(1, Math.round((seg.end - seg.start) / spd * fps)), working: true, stage: null, saveProg: 0 };
  1113→          try {
  1114→            await renderPartWorker(seg, srcUrl, W, H, fps, br, vCfg, aCodec, slot, () => {
  1115→              doneParts += 0.0001;
  1116→              if (ui.prog) ui.prog(clamp((mergeCtx.off + (seg.end - seg.start)) / Math.max(0.1, expLen() || 1) * 0, 0, 1)); /* prog pakai fr di bawah */
  1117→            }, mergeCtx);
  1118→          } catch (e) {
  1119→            if (state.acc === 'gpu' && swCfg && !xAborted()) {
  1120→              console.warn(`merge part gagal (${e && e.message || e}) — ulangi dengan encoder software`);
  1121→              await renderPartWorker(seg, srcUrl, W, H, fps, br, swCfg, aCodec, slot, null, mergeCtx);
  1122→            } else throw e;
  1123→          }
  1124→          mergeCtx.off += (seg.end - seg.start) / spd;
  1125→          if (ui.prog) ui.prog(clamp(mergeCtx.off / Math.max(0.1, (state.duration || 1) * paths.length), 0, 1));
  1126→          if (ui.sub) ui.sub(`[${i + 1}/${paths.length}] ${nmv} · part ${slot.seg.i + 1}/${segs.length} selesai · total ${fmtT(mergeCtx.off)}`);
  1127→        }
  1128→      } finally {
  1129→        if (!state.file.virtual) try { URL.revokeObjectURL(srcUrl); } catch (e) { }
  1130→      }
  1131→    }
  1132→    if (state.abort) throw new Error('Dibatalkan');
  1133→    if (ui.sub) ui.sub('Menuntaskan header MP4…');
  1134→    await withTimeout(mergeCtx.chain, 180000, 'menulis file gabungan macet (cek ruang disk)');
  1135→    if (mergeCtx.wsErr) throw mergeCtx.wsErr;
  1136→    await withTimeout(Promise.resolve().then(() => mergeCtx.muxer.finalize()), 60000, 'penulisan header MP4 gabungan macet');
  1137→    await window.kinostra.endWrite(mergeCtx.wsId); mergeCtx.wsId = null;
  1138→    const stt = await window.kinostra.stat(mergeCtx.outPath);
  1139→    const el = (performance.now() - t0) / 1000;
  1140→    if (ui.sub) ui.sub(`selesai dalam ${el < 90 ? el.toFixed(0) + ' detik' : (el / 60).toFixed(1) + ' menit'}`);
  1141→    return [{ name: nm, path: mergeCtx.outPath, size: stt && stt.ok ? stt.size : mergeCtx.sent }];
  1142→  } catch (e) {
  1143→    /* gagal → buang file setengah jadi */
  1144→    try { if (mergeCtx.wsId) { const id = mergeCtx.wsId; mergeCtx.wsId = null; await window.kinostra.abortWrite(id); } } catch (e2) { }
  1145→    throw e;
  1146→  } finally {
  1147→    state._noPart = false;
  1148→    state.exportFatal = prevFatal;
  1149→  }
  1150→}
