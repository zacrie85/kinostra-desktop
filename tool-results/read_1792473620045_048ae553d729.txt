    50→  let venc = null;
    51→  try {
    52→    const c = document.createElement('canvas'); c.width = W; c.height = H;
    53→    const g = c.getContext('2d');

    56→    venc = new VideoEncoder({
    57→      output: () => chunks++,
    58→      error: e => { err = err || (e && e.message) || 'encoder error'; }
    59→    });
    60→    venc.configure({ ...cfg, latencyMode: 'realtime' });
    61→    const t0 = performance.now();
    62→    for (let i = 0; i < frames; i++) {
    63→      /* konten BERGERAK — encoder bekerja nyata, angka fps jujur */
    64→      g.fillStyle = `hsl(${(i * 12) % 360} 60% ${18 + (i % 26)}%)`;
    65→      g.fillRect(0, 0, W, H);

    69→      venc.encode(vf, { keyFrame: i === 0 }); vf.close();

    71→    await withTimeout(venc.flush().then(() => { flushed = true; }), 10000, 'probe flush macet');

   110→  const base = ['avc1.640028', 'avc1.4D0028', 'avc1.42002A'];
   111→  const mkHw = c => ({ codec: c, width: W, height: H, bitrate: br, framerate: fps, hardwareAcceleration: 'prefer-hardware' });
   112→  const mkSw = c => ({ codec: c, width: W, height: H, bitrate: br, framerate: fps, hardwareAcceleration: 'prefer-software' });
   113→  const key = `${W}x${H}:${br}:${fps}:${state.acc}`;
   114→  if (_accCache && _accCache.key === key) { _lastEnc = _accCache.res.enc; return _accCache.res.cfg; }
   115→
   116→  const probeKind = async (mk, kind) => {
   117→    for (const c of base) {
   118→      try {

   135→      toast('Encoder GPU tidak merespons — otomatis memakai encoder CPU', 'warn');

   146→    const sw = await probeKind(mkSw, 'sw');

   157→      } catch (e) { }
   158→    }
   159→  }
   160→  if (res) {
   161→    _accCache = { key, res };
   162→    _lastEnc = res.enc;
   163→    return res.cfg;
   164→  }
   165→  return null;
   166→}

   218→async function renderMix(start, durOut, spd = 1) {
   219→  const sr = 48000, len = Math.max(1, Math.ceil(durOut * sr));
   220→  const oc = new OfflineAudioContext(2, len, sr);
   221→  /* v3.0: AUDIO SEHAT — rantai perbaikan: filter desis (opsional) →
   222→     normalizer (RMS diukur sekali per video) → kompresor lembut.
   223→     Bila mati, sumber tersambung langsung ke destination (dulu). */
   224→  let sink = null;
   225→  if (state.audioFix && state.audioFix.on) {
   226→    sink = oc.createGain();
   227→    let cur = sink;
   228→    if (state.audioFix.hiss) {
   229→      const hp = oc.createBiquadFilter(); hp.type = 'highpass'; hp.frequency.value = 95; hp.Q.value = 0.6;
   230→      const lp = oc.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 11500;
   231→      cur.connect(hp); hp.connect(lp); cur = lp;
   232→    }
   233→    const norm = oc.createGain();
   234→    const rms = (typeof measureAudioRms === 'function') ? measureAudioRms() : 0;
   235→    norm.gain.value = rms > 0.004 ? clamp(0.16 / rms, 0.5, 4) : 1;
   236→    cur.connect(norm);
   237→    const comp = oc.createDynamicsCompressor();
   238→    comp.threshold.value = -18; comp.knee.value = 20; comp.ratio.value = 2.6;
   239→    comp.attack.value = 0.01; comp.release.value = 0.24;
   240→    norm.connect(comp); comp.connect(oc.destination);
   241→  }
   242→  const connectSrc = node => { if (sink) node.connect(sink); else node.connect(oc.destination); };
   243→  /* v2.10 FIX: dulu s.start(0, start, durBuffer − start) melempar
   244→     InvalidStateError ketika start > durasi buffer (mis. WAKTU MULAI 1:05
   245→     padahal skor musik masih milik video sebelumnya yang lebih pendek) —
   246→     inilah pemicu utama batch ekspor "selalu dibatalkan" sejak v2.9.
   247→     Sekarang: audio asli dilewati bila sudah lewat, musik diputar BERULANG
   248→     (loop) sehingga durasi buffer berapa pun tidak pernah crash.
   249→     v2.12: kecepatan → audio asli diputar lebih cepat/lambat memakai
   250→     playbackRate (pitch ikut naik/turun seperti efek fast-forward klasik). */
   251→  if (state.audioBuffer && state.audioBuffer.duration > 0.01 && start < state.audioBuffer.duration - 0.01) {
   252→    const s = oc.createBufferSource(); s.buffer = state.audioBuffer;
   253→    s.playbackRate.value = spd;
   254→    const g = oc.createGain(); g.gain.value = state.audioGain;
   255→    s.connect(g); connectSrc(g);
   256→    s.start(0, start, Math.max(0.01, Math.min(durOut, (state.audioBuffer.duration - start) / spd)));
   257→  }
   258→  if (state.music.buffer && state.music.buffer.duration > 0.01) {
   259→    const s = oc.createBufferSource(); s.buffer = state.music.buffer;
   260→    s.playbackRate.value = spd;
   261→    const g = oc.createGain(); g.gain.value = state.music.gain;
   262→    s.connect(g); connectSrc(g);
   263→    s.loop = true;   /* buffer lebih pendek dari part → mengulang, bukan crash */
   264→    s.start(0, start % state.music.buffer.duration, durOut);
   265→  }
   266→  /* v3.0: TTS NARASI — suara AI offline diletakkan di waktu media
   267→     state.tts.at; ikut kecepatan video (spd). Potongan di luar part
   268→     dilewati; bagian yang terpotong di awal part disesuaikan. */
   269→  const T = state.tts;
   270→  if (T && T.buffer && T.buffer.duration > 0.01) {
   271→    let outOffN = (T.at - start) / spd, srcOff = 0;
   272→    if (outOffN < 0) { srcOff = -outOffN * spd; outOffN = 0; }
   273→    if (outOffN < durOut - 0.01 && srcOff < T.buffer.duration) {
   274→      const s = oc.createBufferSource(); s.buffer = T.buffer;
   275→      s.playbackRate.value = spd;
   276→      const g = oc.createGain(); g.gain.value = clamp(T.gain == null ? 0.9 : T.gain, 0, 1.6);
   277→      s.connect(g); connectSrc(g);
   278→      s.start(outOffN, srcOff, Math.max(0.01, Math.min(durOut - outOffN, (T.buffer.duration - srcOff) / spd)));
   279→    }
   280→  }
   281→  return oc.startRendering();
   282→}
