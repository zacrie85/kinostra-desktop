    49→async function probeEncoderFPS(cfg, W, H, frames = 30) {
    50→  let venc = null;
    51→  try {
    52→    const c = document.createElement('canvas'); c.width = W; c.height = H;
    53→    const g = c.getContext('2d');
    54→    g.fillStyle = '#1a2a3a'; g.fillRect(0, 0, W, H);
    55→    let chunks = 0, err = null, flushed = false;
    56→    venc = new VideoEncoder({
    57→      output: () => chunks++,
    58→      error: e => { err = err || (e && e.message) || 'encoder error'; }
    59→    });
    60→    venc.configure({ ...cfg, latencyMode: 'realtime' });
    61→    const t0 = performance.now();
    62→    for (let i = 0; i < frames; i++) {
    63→      /* konten BERGERAK — encoder bekerja nyata, angka fps jujur */
    64→      g.fillStyle = `hsl(${(i * 12) % 360} 60% ${18 + (i % 26)}%)`;
    65→      g.fillRect(0, 0, W, H);
    66→      g.fillStyle = '#fff';
    67→      g.fillRect((i * 37) % (W - 130), (i * 53) % (H - 70), 120, 68);
    68→      const vf = new VideoFrame(c, { timestamp: i * 41666, duration: 41666 });
    69→      venc.encode(vf, { keyFrame: i === 0 }); vf.close();
    70→    }
    71→    await withTimeout(venc.flush().then(() => { flushed = true; }), 10000, 'probe flush macet');
    72→    const ms = performance.now() - t0;
    73→    const ok = flushed && chunks >= frames * 0.9 && !err;
    74→    return { ok, fps: ok ? Math.round(frames / Math.max(1, ms) * 1000) : 0 };
    75→  } catch (e) {
    76→    return { ok: false, fps: 0 };
    77→  } finally {
    78→    /* v2.15 FIX BOCOR SESI ENCODER: dulu bila flush probe kena timeout,
    79→       exception melompati venc.close() → sesi encoder Media Foundation
    80→       TETAP TERBUKA di GPU process. Sesi bocor menumpuk sepanjang batch →
    81→       film ke-3 dst encoder hardware menolak/macet → tiap part membayar
    82→       gagal-hardware 30-45 dtk + render ulang software → ekspor 4× lipat.
    83→       Kini close() dijamin jalan di SEMUA jalur (sukses/gagal/timeout). */
    84→    try { if (venc && venc.state !== 'closed') venc.close(); } catch (e) { }
    85→  }
    86→}

   109→async function pickVCodecCfg(W, H, br, fps) {
   110→  const base = ['avc1.640028', 'avc1.4D0028', 'avc1.42002A'];
   111→  const mkHw = c => ({ codec: c, width: W, height: H, bitrate: br, framerate: fps, hardwareAcceleration: 'prefer-hardware' });
   112→  const mkSw = c => ({ codec: c, width: W, height: H, bitrate: br, framerate: fps, hardwareAcceleration: 'prefer-software' });
   113→  const key = `${W}x${H}:${br}:${fps}:${state.acc}`;
   114→  if (_accCache && _accCache.key === key) { _lastEnc = _accCache.res.enc; return _accCache.res.cfg; }
   115→
   116→  const probeKind = async (mk, kind) => {
   117→    for (const c of base) {
   118→      try {
   119→        const cfg = mk(c);
   120→        const s = await VideoEncoder.isConfigSupported(cfg);
   121→        if (!s.supported) continue;
   122→        const r = await probeEncoderFPS(cfg, W, H);
   123→        if (r.ok && r.fps > 0) return { cfg, enc: { kind, fps: r.fps } };
   124→      } catch (e) { }
   125→    }
   126→    return null;
   127→  };
   128→
   129→  let res = null;
   130→  if (state.acc === 'gpu') {
   131→    /* DIPAKSA GPU — bila encoder hardware tidak merespons, jatuh ke CPU
   132→       dengan peringatan (ekspor tetap jalan, bukan gagal) */
   133→    res = await probeKind(mkHw, 'hw');
   134→    if (!res) {
   135→      toast('Encoder GPU tidak merespons — otomatis memakai encoder CPU', 'warn');
   136→      res = await probeKind(mkSw, 'sw');
   137→    }
   138→  } else if (state.acc === 'cpu') {
   139→    res = (await probeKind(mkSw, 'sw')) || (await probeKind(mkHw, 'hw'));
   140→  } else {
   141→    /* OTOMATIS — ukur keduanya, pilih yang TERBUKTI lebih kencang.
   142→       GPU harus lebih cepat ≥15% untuk mengalahkan CPU (stabil & hemat risiko).
   143→       v2.15: bila hardware sudah didemosi (berulang kali macet di film
   144→       sebelumnya), langsung software — tanpa uji GPU lagi. */
   145→    const hw = (state.acc === 'auto' && _hwDemoted) ? null : await probeKind(mkHw, 'hw');
   146→    const sw = await probeKind(mkSw, 'sw');
   147→    res = !sw ? hw : (!hw ? sw : (hw.enc.fps >= sw.enc.fps * 1.15 ? hw : sw));
   148→  }
   149→  if (!res) {
   150→    /* upaya terakhir: tanpa preferensi hardware */
   151→    for (const c of base) {
   152→      try {
   153→        const cfg = { codec: c, width: W, height: H, bitrate: br, framerate: fps };
   154→        const s = await VideoEncoder.isConfigSupported(cfg);
   155→        const r = s.supported ? await probeEncoderFPS(cfg, W, H) : { ok: false };
   156→        if (r.ok) { res = { cfg, enc: { kind: 'sw', fps: r.fps } }; break; }
   157→      } catch (e) { }
   158→    }
   159→  }
   160→  if (res) {
   161→    _accCache = { key, res };
   162→    _lastEnc = res.enc;
   163→    return res.cfg;
   164→  }
   165→  return null;
   166→}

   211→  return new Promise((res, rej) => {
   212→    const tm = setTimeout(() => rej(new Error(msg)), ms);
   213→    Promise.resolve(p).then(
   214→      v => { clearTimeout(tm); res(v); },

   218→async function renderMix(start, durOut, spd = 1) {
   219→  const sr = 48000, len = Math.max(1, Math.ceil(durOut * sr));
   220→  const oc = new OfflineAudioContext(2, len, sr);
   221→  /* v3.0: AUDIO SEHAT — rantai perbaikan: filter desis (opsional) →
   222→     normalizer (RMS diukur sekali per video) → kompresor lembut.
   223→     Bila mati, sumber tersambung langsung ke destination (dulu). */
   224→  let sink = null;
   225→  if (state.audioFix && state.audioFix.on) {
   226→    sink = oc.createGain();
   227→    let cur = sink;
   228→    if (state.audioFix.hiss) {
   229→      const hp = oc.createBiquadFilter(); hp.type = 'highpass'; hp.frequency.value = 95; hp.Q.value = 0.6;
   230→      const lp = oc.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 11500;
   231→      cur.connect(hp); hp.connect(lp); cur = lp;
   232→    }
   233→    const norm = oc.createGain();
   234→    const rms = (typeof measureAudioRms === 'function') ? measureAudioRms() : 0;
   235→    norm.gain.value = rms > 0.004 ? clamp(0.16 / rms, 0.5, 4) : 1;
   236→    cur.connect(norm);
   237→    const comp = oc.createDynamicsCompressor();
   238→    comp.threshold.value = -18; comp.knee.value = 20; comp.ratio.value = 2.6;
   239→    comp.attack.value = 0.01; comp.release.value = 0.24;
   240→    norm.connect(comp); comp.connect(oc.destination);
   241→  }
   242→  const connectSrc = node => { if (sink) node.connect(sink); else node.connect(oc.destination); };
   243→  /* v2.10 FIX: dulu s.start(0, start, durBuffer − start) melempar
   244→     InvalidStateError ketika start > durasi buffer (mis. WAKTU MULAI 1:05
   245→     padahal skor musik masih milik video sebelumnya yang lebih pendek) —
   246→     inilah pemicu utama batch ekspor "selalu dibatalkan" sejak v2.9.
   247→     Sekarang: audio asli dilewati bila sudah lewat, musik diputar BERULANG
   248→     (loop) sehingga durasi buffer berapa pun tidak pernah crash.
   249→     v2.12: kecepatan → audio asli diputar lebih cepat/lambat memakai
   250→     playbackRate (pitch ikut naik/turun seperti efek fast-forward klasik). */
   251→  if (state.audioBuffer && state.audioBuffer.duration > 0.01 && start < state.audioBuffer.duration - 0.01) {
   252→    const s = oc.createBufferSource(); s.buffer = state.audioBuffer;
   253→    s.playbackRate.value = spd;
   254→    const g = oc.createGain(); g.gain.value = state.audioGain;
   255→    s.connect(g); connectSrc(g);
   256→    s.start(0, start, Math.max(0.01, Math.min(durOut, (state.audioBuffer.duration - start) / spd)));
   257→  }
   258→  if (state.music.buffer && state.music.buffer.duration > 0.01) {
   259→    const s = oc.createBufferSource(); s.buffer = state.music.buffer;
   260→    s.playbackRate.value = spd;
   261→    const g = oc.createGain(); g.gain.value = state.music.gain;
   262→    s.connect(g); connectSrc(g);
   263→    s.loop = true;   /* buffer lebih pendek dari part → mengulang, bukan crash */
   264→    s.start(0, start % state.music.buffer.duration, durOut);
   265→  }
   266→  /* v3.0: TTS NARASI — suara AI offline diletakkan di waktu media
   267→     state.tts.at; ikut kecepatan video (spd). Potongan di luar part
   268→     dilewati; bagian yang terpotong di awal part disesuaikan. */
   269→  const T = state.tts;
   270→  if (T && T.buffer && T.buffer.duration > 0.01) {
   271→    let outOffN = (T.at - start) / spd, srcOff = 0;
   272→    if (outOffN < 0) { srcOff = -outOffN * spd; outOffN = 0; }
   273→    if (outOffN < durOut - 0.01 && srcOff < T.buffer.duration) {
   274→      const s = oc.createBufferSource(); s.buffer = T.buffer;
   275→      s.playbackRate.value = spd;
   276→      const g = oc.createGain(); g.gain.value = clamp(T.gain == null ? 0.9 : T.gain, 0, 1.6);
   277→      s.connect(g); connectSrc(g);
   278→      s.start(outOffN, srcOff, Math.max(0.01, Math.min(durOut - outOffN, (T.buffer.duration - srcOff) / spd)));
   279→    }
   280→  }
   281→  return oc.startRendering();
   282→}
   283→/* ---------- ENCODING AUDIO (v2.15: PER BLOK 60 dtk — RAM tetap datar) ----------
   284→   Dulu: SELURUH audio part dirender dulu oleh OfflineAudioContext →
   285→   AudioBuffer sebesar durasi part (film 60 mnt ≈ 700 MB Float32 di RAM
   286→   per part!) baru di-encode. Kini: blok 60 dtk dirender & di-encode
   287→   bergantian. Hasil identik — offset sumber & posisi loop musik kontinu
   288→   antar blok (blok k = waktu output [k·60, (k+1)·60) → sumber mulai
   289→   seg.start + outOff·spd), stempel waktu dihitung dari penghitung global.
   290→   v3.0: mergeOff = offset waktu output global saat GABUNG VIDEO;
   291→   fbuf dipakai ulang antar detik (kurangi tekanan GC ±40%). */
   292→async function encodeAudioStream(muxer, seg, durOut, spd = 1, codec, mergeOff = 0) {
   293→  const sr = 48000, chs = 2;
   294→  const aenc = new AudioEncoder({ output: (c, m) => muxer.addAudioChunk(c, m), error: e => console.warn('aenc', e) });
   295→  aenc.configure({ codec, sampleRate: sr, numberOfChannels: chs, bitrate: 160000 });
   296→  try {
   297→    const BLK = 60;
   298→    const fbuf = new Float32Array(sr * chs);
   299→    let outOff = 0;
   300→    while (outOff < durOut - 1e-6) {
   301→      if (state.abort) throw new Error('Dibatalkan');
   302→      if (state.exportFatal) throw state.exportFatal;
   303→      const len = Math.min(BLK, durOut - outOff);
   304→      const mix = await renderMix(seg.start + outOff * spd, len, spd);
   305→      const L = mix.getChannelData(0), R2 = chs > 1 ? mix.getChannelData(1) : L;
   306→      for (let off = 0; off < mix.length; off += sr) {
   307→        if (state.abort) throw new Error('Dibatalkan');
   308→        if (state.exportFatal) throw state.exportFatal;
   309→        const n = Math.min(sr, mix.length - off);
   310→        fbuf.set(L.subarray(off, off + n), 0);
   311→        if (chs > 1) fbuf.set(R2.subarray(off, off + n), n);
   312→        const ad = new AudioData({ format: 'f32-planar', sampleRate: sr, numberOfFrames: n, numberOfChannels: chs,
   313→          timestamp: Math.round((mergeOff + outOff + off / sr) * 1e6), data: fbuf });
   314→        aenc.encode(ad); ad.close();
   315→        let drainMs = 0;
   316→        while (aenc.encodeQueueSize > 10) { await sleep(2); if (state.abort) throw new Error('Dibatalkan'); if (state.exportFatal) throw state.exportFatal; if ((drainMs += 2) > 30000) throw new Error('encoder audio macet'); }
   317→      }
   318→      outOff += len;
   319→    }
   320→    await aenc.flush();
   321→  } finally {
   322→    try { aenc.close(); } catch (e) { }   /* v2.15: sesi encoder audio selalu ditutup */
   323→  }
   324→}

   357→async function renderFramesBySeek(v, ecv, ec, venc, seg, fps, slot, doneCb, noSeek = false, spd = 1, tsOff = 0) {
   358→  const n = Math.max(1, Math.round((seg.end - seg.start) / spd * fps));
   359→  let seekP = null;
   360→  for (let i = 0; i < n; i++) {
   361→    if (state.abort) throw new Error('Dibatalkan');
   362→    if (state.exportFatal) throw state.exportFatal;
   363→    if (seekP) { await seekP; seekP = null; }
   364→    const t = Math.min(seg.start + i * spd / fps, Math.max(0, (v.duration || seg.end) - 0.011));
   365→    setCompSrc(v);
   366→    drawComposition(ec, ecv.width, ecv.height, t);
   367→    setCompSrc(null);
   368→    const vf = new VideoFrame(ecv, { timestamp: Math.round(tsOff + i * 1e6 / fps), duration: Math.round(1e6 / fps) });
   369→    venc.encode(vf, { keyFrame: i % (fps * 2) === 0 }); vf.close();
   370→    if (!noSeek && i + 1 < n) seekP = seekTo(seg.start + (i + 1) * spd / fps, v);
   371→    while (venc.encodeQueueSize > 14) await sleep(2);
   372→    slot.fr = i + 1;
   373→    doneCb && doneCb();
   374→  }
   375→  return n;
   376→}

   378→/* ---------- WORKER SATU PART (v2.2 TURBO) ----------
   379→   Elemen video sendiri + canvas sendiri + muxer sendiri.
   380→   Frame diambil sambil video diputar (tanpa seek per frame). */
   381→async function renderPartWorker(seg, srcUrl, W, H, fps, br, vCfg, aCodec, slot, doneCb, mergeCtx = null) {
   382→  /* v2.12: KECEPATAN video — 1 = normal, 2 = 2× cepat (part ½ lebih pendek),
   383→     0.5 = slow-motion. Durasi output = durasi sumber ÷ spd. */
   384→  const spd = clamp(state.speed || 1, 0.25, 4);
   385→  const durS = (seg.end - seg.start) / spd;
   386→  const v = document.createElement('video');
   387→  v.muted = true; v.playsInline = true; v.preload = 'auto';
   388→  v.style.cssText = 'position:fixed;left:-9999px;top:-9999px;width:2px;height:2px;opacity:0';
   389→  v.src = srcUrl;
   390→  document.body.appendChild(v);
   391→  v.addEventListener('error', () => console.error(`part ${seg.i + 1} media error: code=${v.error && v.error.code} msg=${v.error && v.error.message}`));
   392→  let venc = null, _stage = 'init', n = 0;
   393→  let audioP = null, audioErr = null;   /* v3.0: audio paralel dgn video */
   394→  try {
   395→    await new Promise((res, rej) => {
   396→      const ok = () => { clean(); res(); };
   397→      const bad = () => { clean(); rej(new Error('worker: meta tidak termuat')); };
   398→      const clean = () => { v.removeEventListener('loadedmetadata', ok); v.removeEventListener('error', bad); };
   399→      v.addEventListener('loadedmetadata', ok); v.addEventListener('error', bad);
   400→      setTimeout(() => { if (v.readyState >= 1) ok(); }, 6000);
   401→    });
   402→    _stage = 'seek0';
   403→    await seekTo(seg.start, v);
   404→
   405→    _stage = 'canvas';
   406→    const ecv = document.createElement('canvas'); ecv.width = W; ecv.height = H;
   407→    const ec = ecv.getContext('2d', { alpha: false });
   408→    _stage = 'muxer';
   409→    /* v2.15: part BESAR ditulis LANGSUNG ke disk (StreamTarget → IPC 8 MB) —
   410→       dulu SELURUH berkas MP4 menumpuk di RAM (ArrayBufferTarget) sampai
   411→       render selesai; part 30-60 mnt = gigabyte per part di memori,
   412→       batch film panjang lalu kehabisan memori → ekspor merambat.
   413→       Part kecil tetap in-memory (fastStart penuh, kompatibilitas maks).
   414→       v3.0: GABUNG VIDEO — semua part memakai SATU muxer bersama milik
   415→       driver merge (satu file MP4 kontinu di disk); worker tidak menulis
   416→       file sendiri, tidak finalize — hanya meng-encode. */
   417→    const nm = partFileName(state.title, state.partPrefix, seg.i + 1);
   418→    const estBytes = Math.round(durS * (br + (aCodec ? 160000 : 0)) / 8) + 4e6;
   419→    const streamDisk = !mergeCtx && estBytes > (state._streamThreshMB || 384) * 1048576;
   420→    let wsId = null, wsErr = null, wsPath = mergeCtx ? mergeCtx.outPath : nm, chain = Promise.resolve(), sent = 0;
   421→    const mkAudio = aCodec ? { audio: { codec: aCodec === 'mp4a.40.2' ? 'aac' : 'opus', numberOfChannels: 2, sampleRate: 48000 } } : {};
   422→    const mkVideo = { codec: 'avc', width: W, height: H, frameRate: fps };
   423→    let muxer;
   424→    if (mergeCtx) {
   425→      muxer = mergeCtx.muxer;
   426→      wsErr = mergeCtx.wsErr;
   427→    } else if (streamDisk) {
   428→      const w = await window.kinostra.beginWrite(slot.dir, nm);
   429→      wsId = w.id; wsPath = w.path;
   430→      console.info(`part ${seg.i + 1}: muxer streaming disk aktif (±${fmtMB(estBytes)}) — RAM tetap datar (v2.15)`);
   431→      muxer = new Mp4Muxer.Muxer({
   432→        target: new Mp4Muxer.StreamTarget({
   433→          chunked: true, chunkSize: 8 * 1024 * 1024,
   434→          /* onData(data, posisi) — tulis-acak ke disk via pwrite (IPC);
   435→             rantai `chain` menjaga urutan panggilan, patch header mdat di
   436→             posisi lama oleh mp4-muxer (fastStart:false) tetap benar */
   437→          onData: (data, pos) => {
   438→            if (wsErr) return;
   439→            sent += data.byteLength;
   440→            slot.saveProg = Math.min(0.98, sent / estBytes);
   441→            chain = chain.then(() => window.kinostra.writeChunkAt(wsId, data, pos)).catch(e => { wsErr = wsErr || e; });
   442→          }
   443→        }),
   444→        fastStart: false, firstTimestampBehavior: 'offset',
   445→        video: mkVideo, ...mkAudio
   446→      });
   447→    } else {
   448→      muxer = new Mp4Muxer.Muxer({
   449→        target: new Mp4Muxer.ArrayBufferTarget(), fastStart: 'in-memory', firstTimestampBehavior: 'offset',
   450→        video: mkVideo, ...mkAudio
   451→      });
   452→    }
   453→    if (aCodec) {
   454→      _stage = 'audio';
   455→      /* v2.15: encode audio PER BLOK — RAM datar di part panjang
   456→         v3.0 OPTIMASI KECEPATAN: audio di-encode BERSAMAAN dengan render
   457→         video (dulu audio selesai dulu → baru video: waktu menjumlah).
   458→         Kedua encoder mengalir ke muxer yang sama — mp4-muxer menerima
   459→         potongan video/audio ter-selang; hasil identik, ±30-45% lebih cepat. */
   460→      audioP = withTimeout(
   461→        encodeAudioStream(muxer, seg, durS, spd, aCodec, mergeCtx ? mergeCtx.off : 0),
   462→        300000, 'encoding audio macet'
   463→      ).catch(e => { audioErr = audioErr || e; });
   464→    }
   465→    _stage = 'configure';
   466→    const hw = vCfg.hardwareAcceleration === 'prefer-hardware';
   467→    /* v2.7 FIX "SUARA JALAN, GAMBAR BEKU": encoder GPU Windows (Media
   468→       Foundation) dengan latencyMode 'quality' dapat mengeluarkan chunk
   469→       B-frame berurutan DECODE (timestamp bolak-balik). mp4-muxer melempar
   470→       error monotonic DI DALAM callback output → pengecualian tertelan
   471→       diam-diam → hanya beberapa frame pertama masuk MP4 → gambar beku +
   472→       audio normal + ekspor "berhasil". Perisai 4 lapis:
   473→       1) latencyMode 'realtime' utk SEMUA encoder (tanpa reorder B-frame)
   474→       2) timestamp turun > 5 ms terdeteksi → gagal cepat → auto-retry SW
   475→       3) error encoder diteruskan ke promise (bukan hanya console)
   476→       4) verifikasi jumlah chunk video setelah flush (≥ 90% frame) */
   477→    let vChunks = 0, lastVts = -Infinity, vencErrMsg = null;
   478→    venc = new VideoEncoder({
   479→      output: (c, m) => {
   480→        const ts = typeof c.timestamp === 'number' ? c.timestamp : 0;
   481→        if (vChunks > 0 && ts <= lastVts) {
   482→          if (lastVts - ts > 5000) {
   483→            vencErrMsg = `encoder mengeluarkan frame dengan urutan waktu tidak naik (${Math.round(lastVts / 1000)} → ${Math.round(ts / 1000)} ms)`;
   484→            console.error('venc-stream', vencErrMsg);
   485→          }
   486→          return; /* jangan masukkan chunk rusak/nol-durasi ke muxer */
   487→        }
   488→        lastVts = ts; vChunks++;
   489→        muxer.addVideoChunk(c, m);
   490→      },
   491→      error: e => { vencErrMsg = (e && e.message) || String(e); console.error('venc', vencErrMsg); }
   492→    });
   493→    venc.configure({ ...vCfg, latencyMode: 'realtime' });
   494→    const CAP = hw ? 40 : 14;
   495→
   496→    /* gambar 1 frame pada waktu media m → encode dengan timestamp nyata
   497→       v2.12: PTS = waktu OUTPUT = (m − mulai part) ÷ kecepatan
   498→       v3.0: + offset global mergeCtx.off saat GABUNG VIDEO */
   499→    const drawAt = (t, ts) => {
   500→      setCompSrc(v);
   501→      drawComposition(ec, W, H, t);
   502→      setCompSrc(null);
   503→      const vf = new VideoFrame(ecv, { timestamp: ts, duration: Math.round(1e6 / fps) });
   504→      venc.encode(vf, { keyFrame: n % (fps * 2) === 0 }); vf.close();
   505→      n++; slot.fr = n;
   506→    };
   507→    const outTs = (m) => Math.max(0, Math.round((((mergeCtx ? mergeCtx.off : 0) + (m - seg.start) / spd) * 1e6)));
   508→
   509→    if (RVFC_OK && !state.isAudio) {
   510→      /* ============ v2.2: CAPTURE PLAYBACK (tanpa seek per frame) ============
   511→         v2.5: + WATCHDOG ANTI-MACET. Penyebab lama ekspor "macet di bagian
   512→         akhir tanpa pesan": frame callback (rVFC) berhenti memicu di frame
   513→         terakhir & event ended tidak sampai, atau encoder GPU berhenti
   514→         mengeluarkan output → promise render tidak pernah selesai.
   515→         Sekarang interval 250 ms memantau: video selesai? waktu media
   516→         sudah mencapai akhir part? video error? encoder diam terlalu
   517→         lama? → semua diarahkan SELESAI atau GAGAL DENGAN PESAN JELAS. */
   518→      _stage = 'stream';
   519→      /* v2.11 FIX "KADANG GERAK KADANG BEKU": dulu capRate mulai 2× dan bisa
   520→         naik sampai 4× — di atas kemampuan decoder/compositor, frame sumber
   521→         dibuang beramai-ramai, mediaTime antar frame tersaji LOMPAT, PTS
   522→         ekspor ikut bolong → pemutar menahan frame lama berdetik-detik.
   523→         Sekarang: mulai 1.5×, turun SEGERA saat ada frame hilang, dan setiap
   524→         lompatan DIISI ULANG dengan seek presisi (refillGap) sehingga aliran
   525→         PTS selalu rapat & mulus.
   526→         v2.14: bila encoder GPU (hardware) aktif, langit-langit dinaikkan
   527→         ke 3× — encoding bukan lagi hambatan, decoder-lah penentu. */
   528→      const RATE_MAX = hw ? 3 : (_xPar >= 3 ? 2 : 2.5);
   529→      /* v2.12: toleransi & target lompatan diskalakan kecepatan —
   530→         pada spd 2× jarak sumber antar frame tampil 2× lebih renggang */
   531→      const frameGapTol = 1.8 / fps * spd;
   532→      let capRate = 1.5, dAcc = 0, dN = 0, lastM = -1, hadFirst = false;
   533→      const applyRate = () => { try { v.playbackRate = Math.min(4.2, capRate * spd); } catch (e) { } };
   534→      applyRate();
   535→      let lastAct = Date.now(), qStallSince = 0;
   536→      let refilling = false, refillN = 0, maxGapMs = 0;
   537→      /* v2.11: isi ulang frame yang hilang di interval (from, to) — seek
   538→         presisi slot per slot, encode dengan PTS aslinya. Dipanggil SEBELUM
   539→         frame@to di-encode agar urutan PTS tetap naik. lastAct disegarkan
   540→         tiap slot supaya watchdog 30 dtk tidak mengira render diam. */
   541→      const refillGap = async (from, to) => {
   542→        const step = spd / fps;
   543→        let t = from + step, k = 0;
   544→        while (t < to - step * 0.5) {
   545→          if (state.abort) throw new Error('Dibatalkan');
   546→          if (state.exportFatal) throw state.exportFatal;
   547→          await seekTo(Math.min(t, Math.max(0, (v.duration || seg.end) - 0.011)), v);
   548→          drawAt(t, outTs(t));
   549→          lastAct = Date.now(); k++; t += step;
   550→          doneCb && doneCb();
   551→        }
   552→        refillN += k; return k;
   553→      };
   554→      await new Promise((resolve, rej) => {
   555→        let finished = false;
   556→        const stop = () => { try { v.pause(); } catch (e) { } };
   557→        const fin = () => { if (finished) return; finished = true; stop(); clearInterval(watch); resolve(); };
   558→        const fail = (err) => { if (finished) return; finished = true; stop(); clearInterval(watch); rej(err); };
   559→        /* --- v2.5: WATCHDOG (detak 250 ms) --- */
   560→        const watch = setInterval(() => {
   561→          if (finished) { clearInterval(watch); return; }
   562→          if (state.abort) return fail(new Error('Dibatalkan'));
   563→          if (state.exportFatal) return fail(state.exportFatal);   /* v2.10: part lain sudah gagal */
   564→          /* v2.7: error encoder terdeteksi → gagal cepat → retry software */
   565→          if (vencErrMsg) return fail(new Error('encoder video bermasalah: ' + vencErrMsg));
   566→          if (v.error) return fail(new Error('video rusak/tak terbaca: ' + (v.error.message || ('kode ' + v.error.code))));
   567→          /* KUNCI FIX: tangani akhir video walau frame callback mati —
   568→             inilah yang dulu bikin macet selamanya di bagian akhir */
   569→          if (v.ended || v.currentTime >= seg.end - 0.004) return fin();
   570→          const now = Date.now();
   571→          if (venc.encodeQueueSize > CAP) {
   572→            if (!qStallSince) qStallSince = now;
   573→            else if (now - qStallSince > 45000)
   574→              return fail(new Error('encoder video macet (antrean tidak mengalir 45 dtk) — akan dicoba ulang otomatis dengan encoder software'));
   575→          } else qStallSince = 0;
   576→          if (now - lastAct > 30000)
   577→            return fail(new Error('render diam total 30 dtk — akan dicoba ulang otomatis dengan encoder software'));
   578→          /* dorong ulang bila video diam padahal encoder tidak penuh */
   579→          if (v.paused && !refilling && !v.seeking && venc.encodeQueueSize <= CAP && !v.ended) {
   580→            try { const p = v.play(); if (p && p.catch) p.catch(err => fail(new Error('video tidak bisa dilanjut: ' + (err.message || err)))); }
   581→            catch (err) { fail(new Error('video tidak bisa dilanjut: ' + (err.message || err))); }
   582→          }
   583→        }, 250);
   584→        const onFrame = async (now, meta) => {
   585→          try {
   586→            if (finished) return;
   587→            lastAct = Date.now();
   588→            if (state.abort) return fail(new Error('Dibatalkan'));
   589→            if (state.exportFatal) return fail(state.exportFatal);   /* v2.10 */
   590→            const m = meta.mediaTime;
   591→            if (v.ended || m >= seg.end - 0.004) return fin();
   592→            if (m >= seg.start - 0.06 && m > lastM) {
   593→              /* v2.11: DETEKSI FRAME HILANG — lompatan mediaTime > 1.8× durasi
   594→                 frame berarti decoder/compositor membuang frame. Isi slot yang
   595→                 bolong lewat seek presisi SEBELUM frame m di-encode supaya
   596→                 urutan PTS rapat (tanpa durasi sampel multi-detik = beku). */
   597→              const gap = hadFirst ? m - lastM : 0;
   598→              if (gap > frameGapTol) {
   599→                maxGapMs = Math.max(maxGapMs, gap * 1000);
   600→                refilling = true; stop();
   601→                const k = await refillGap(lastM, m);
   602→                if (finished) return;
   603→                await seekTo(Math.min(m, Math.max(0, (v.duration || seg.end) - 0.011)), v);
   604→                refilling = false;
   605→                if (finished) return;
   606→                /* decoder kewalahan pada rate ini → turun SEGERA */
   607→                if (capRate > 1) { capRate = Math.max(1, capRate - 0.5); applyRate(); }
   608→                dAcc = 0; dN = 0;
   609→              }
   610→              drawAt(m, outTs(m));
   611→              hadFirst = true;
   612→              /* rate adaptif: naik HANYA bila jendela bersih tanpa lompatan
   613→                 v2.12: target jarak antar frame = spd/fps (waktu sumber) */
   614→              if (lastM >= 0) {
   615→                const d = m - lastM;
   616→                if (d > 0 && d < 0.6 * spd) {
   617→                  dAcc += d; dN++;
   618→                  if (dN >= 24) {
   619→                    const avg = dAcc / dN; dAcc = 0; dN = 0; const tg = spd / fps;
   620→                    if (avg > tg * 1.5 && capRate > 1) { capRate = Math.max(1, capRate - 0.5); applyRate(); }
   621→                    else if (avg < tg * 1.18 && capRate < RATE_MAX) { capRate = Math.min(RATE_MAX, capRate + 0.5); applyRate(); }
   622→                  }
   623→                }
   624→              }
   625→              lastM = m;
   626→              doneCb && doneCb();
   627→            }
   628→            /* backpressure: pause bila encoder tertinggal, lanjut otomatis */
   629→            if (venc.encodeQueueSize > CAP) { stop(); setTimeout(pump, 5); return; }
   630→            pump();
   631→          } catch (e) {
   632→            refilling = false;
   633→            fail(e && e.message ? e : new Error(String(e)));
   634→          }
   635→        };
   636→        const pump = () => {
   637→          if (finished) return;
   638→          if (state.abort) return fail(new Error('Dibatalkan'));
   639→          if (state.exportFatal) return fail(state.exportFatal);   /* v2.10 */
   640→          if (v.ended || lastM >= seg.end - 0.004) return fin();
   641→          if (v.paused) {
   642→            try { const p = v.play(); if (p && p.catch) p.catch(err => fail(new Error('video tidak bisa diputar: ' + (err.message || err)))); }
   643→            catch (err) { return fail(new Error('video tidak bisa diputar: ' + (err.message || err))); }
   644→          }
   645→          try { v.requestVideoFrameCallback(onFrame); } catch (e) { fin(); }
   646→        };
   647→        v.addEventListener('ended', fin, { once: true });
   648→        v.addEventListener('error', () => fail(new Error('video error saat dirender')), { once: true });
   649→        pump();
   650→      });
   651→      if (refillN) console.info(`part ${seg.i + 1}: ${n} frame dirender · ${refillN} diisi-ulang (lompatan maks ${Math.round(maxGapMs)} ms) — anti-beku v2.11 aktif`);
   652→      if (n === 0) {
   653→        /* sumber aneh / tidak menghasilkan frame — jalankan jalur lama */
   654→        _stage = 'fallback-seek';
   655→        await renderFramesBySeek(v, ecv, ec, venc, seg, fps, slot, doneCb, false, spd, mergeCtx ? mergeCtx.off : 0);
   656→        n = slot.fr;
   657→      }
   658→    } else {
   659→      /* audio-only / tanpa rVFC: jalur frame loop (tanpa seek utk audio) */
   660→      _stage = 'frames';
   661→      await renderFramesBySeek(v, ecv, ec, venc, seg, fps, slot, doneCb, !!state.isAudio, spd, mergeCtx ? mergeCtx.off : 0);
   662→      n = slot.fr;
   663→    }
   664→
   665→    _stage = 'flush';
   666→    slot.stage = 'flush';
   667→    /* v3.0: tunggu encoder audio paralel selesai sebelum flush video */
   668→    if (audioP) { await audioP; if (audioErr) throw audioErr; }
   669→    /* v2.5: flush dengan batas waktu — encoder yang hang tidak lagi
   670→       membuat ekspor macet selamanya tanpa pesan */
   671→    await withTimeout(venc.flush(), 60000, 'encoder video tidak selesai (flush macet 60 dtk)');
   672→    venc.close(); venc = null;
   673→    /* v2.7: VERIFIKASI ALIRAN VIDEO — tangkap dua mode kegagalan encoder
   674→       GPU yang "diam": (a) error/reorder terdeteksi saat merender,
   675→       (b) output kurang dari 90% frame yang dirender. Tanpa ini MP4 tetap
   676→       tersimpan tapi isinya gambar beku. Gagal di sini → auto-retry dengan
   677→       encoder software oleh exportPartsToDir. */
   678→    if (vencErrMsg) throw new Error('aliran video tidak valid: ' + vencErrMsg);
   679→    /* v2.11: verifikasi diperketat 90% → 98% — encoder yang membuang output
   680→       di tengah stream kini terdeteksi → auto-retry software, bukan MP4 beku */
   681→    const minChunks = Math.max(1, Math.ceil(n * 0.98));
   682→    if (vChunks < minChunks)
   683→      throw new Error(`encoder hanya mengeluarkan ${vChunks} dari ${n} frame (gambar akan beku)`);
   684→    _stage = 'finalize';
   685→    slot.stage = 'finalize';
   686→    if (wsErr) throw wsErr;
   687→    if (mergeCtx && mergeCtx.wsErr) throw mergeCtx.wsErr;
   688→    if (mergeCtx) {
   689→      /* v3.0: part gabung — driver merge yang mem-finalize & menyimpan */
   690→      slot.prog = 1; slot.fr = slot.total = Math.max(slot.total, n);
   691→      slot.saveProg = 1;
   692→      return { name: mergeCtx.outName, path: mergeCtx.outPath, size: 0, merged: true };
   693→    }
   694→    await withTimeout(Promise.resolve().then(() => muxer.finalize()), 30000, 'penulisan header MP4 macet');
   695→    _stage = 'save';
   696→    slot.stage = 'save';
   697→    let wr;
   698→    if (streamDisk) {
   699→      /* v2.15: semua byte sudah mengalir ke disk SELAMA render — tinggal
   700→         menuntaskan rantai tulis, menutup stream, membaca ukuran berkas */
   701→      if (wsErr) throw wsErr;
   702→      await withTimeout(chain, 180000, 'menulis file ke disk macet (cek ruang kosong disk)');
   703→      if (wsId) { const id = wsId; wsId = null; await window.kinostra.endWrite(id); }
   704→      const stt = await window.kinostra.stat(wsPath);
   705→      wr = { path: wsPath, size: (stt && stt.ok) ? stt.size : sent };
   706→    } else {
   707→      wr = await withTimeout(
   708→        saveBufferToDir(muxer.target.buffer, slot.dir, nm, (w, t) => { slot.saveProg = w / t; }),
   709→        180000, 'menyimpan file ke disk macet (cek ruang kosong disk)');
   710→    }
   711→    slot.prog = 1; slot.fr = slot.total = Math.max(slot.total, n);
   712→    return { name: wr.path.split(/[\\/]/).pop(), path: wr.path, size: wr.size };
   713→  } catch (err) {
   714→    if (_stage !== 'frames' && _stage !== 'stream') console.error(`part ${seg.i + 1} GAGAL di tahap ${_stage}: ${err && err.name || 'ERR'} | ${err && err.message || err}`);
   715→    throw err;
   716→  } finally {
   717→    try { if (venc && venc.state !== 'closed') venc.close(); } catch (e) { }
   718→    /* v2.15: bila gagal di tengah render streaming → buang berkas setengah jadi */
   719→    try { if (wsId) { const id = wsId; wsId = null; window.kinostra.abortWrite(id).catch(() => { }); } } catch (e) { }
   720→    try { v.pause(); v.removeAttribute('src'); v.load(); } catch (e) { }
   721→    try { v.remove(); } catch (e) { }
   722→  }

   725→/* ---------- INTI RENDER: SEMUA PART PARALEL -> folder dir ----------
   726→   ui: {title(segs), prog(p), sub(s), canceled()} */
   727→async function exportPartsToDir(dir, ui) {
   728→  await ensureFontsReady();
   729→  /* v2.6: pastikan decode audio asli (yang kini berjalan di belakang saat
   730→     loading) sudah selesai sebelum menentukan ada tidaknya track audio */
   731→  if (state.audioReady) { try { await state.audioReady; } catch (e) { } }
   732→  const fps = state.fps, sc = parseFloat(state.scale), q = QUAL[state.quality];
   733→  /* v2.12: kecepatan video global — durasi output = sumber ÷ spd */
   734→  const spd = clamp(state.speed || 1, 0.25, 4);
   735→  const dims = outDims(sc), { W, H } = dims;
   736→  const br = Math.round(q.br * (sc === 1 ? 1 : sc === 0.75 ? 0.62 : 0.38));
   737→  /* v2.14: pilih encoder sesuai AKSELERASI (auto = uji kecepatan GPU vs CPU) */
   738→  const vCfg = await pickVCodecCfg(W, H, br, fps);
   739→  if (!vCfg) throw new Error('Tidak ada codec video yang didukung engine');
   740→  const enc = _lastEnc;
   741→  /* v2.5: cadangan encoder SOFTWARE untuk percobaan ulang otomatis */
   742→  const swCfg = await pickSWCfg(W, H, br, fps);
   743→  const wantAudio = !!(state.audioBuffer || state.music.buffer);
   744→  const aCodec = wantAudio ? await pickACodec() : null;
   745→  if (wantAudio && !aCodec) toast('Encoder audio tak tersedia — ekspor tanpa audio', 'warn');
   746→  const segs = segments();
   747→  const par = pickParallelCount(segs.length, enc.kind === 'hw');
   748→  _xPar = par;   /* v2.11: langit-langit playbackRate capture menyesuaikan beban paralel */
   749→  const hw = vCfg.hardwareAcceleration === 'prefer-hardware';
   750→
   751→  /* slot progres per part — total frame = durasi output × fps */
   752→  const slots = segs.map(s => ({ seg: s, dir, prog: 0, fr: 0, total: Math.max(1, Math.round((s.end - s.start) / spd * fps)), working: false, stage: null, saveProg: 0 }));
   753→  const totalFr = slots.reduce((a, s) => a + s.total, 0);
   754→  const t0 = performance.now();
   755→  let lastUi = 0;
   756→  const uiTick = (force) => {
   757→    const now = performance.now();
   758→    if (!force && now - lastUi < 120) return;
   759→    lastUi = now;
   760→    const done = slots.reduce((a, s) => a + Math.min(s.fr, s.total), 0);
   761→    if (ui.prog) ui.prog(done / totalFr);
   762→    if (ui.sub) {
   763→      const el = Math.max(0.001, (now - t0) / 1000);
   764→      const rt = (done / fps / el);
   765→      const parts = slots.map(s => {
   766→        const pct = Math.round(100 * Math.min(s.fr, s.total) / s.total);
   767→        const stg = s.stage === 'flush' ? '· FLUSH' : s.stage === 'finalize' ? '· FINAL' : s.stage === 'save' ? '· SIMPAN' : '';
   768→        const tag = s.retried && s.working ? ' (ULANGI-SW)' : '';
   769→        return s.working ? `P${String(s.seg.i + 1).padStart(2, '0')} ${pct}%${tag}${stg}` : null;
   770→      }).filter(Boolean).join(' · ');
   771→      ui.sub(`${parts || 'menunggu…'}${rt > 0.05 ? ` · ${rt.toFixed(1)}× realtime` : ''}`);
   772→    }
   773→  };
   774→
   775→  /* v2.6: file virtual (impor/batch streaming) — worker memutar langsung
   776→     dari disk via kfile:// (Range), tanpa menyalin file ke memori
   777→     v3.0: dibuat LAZY — LANJUTKAN EKSPOR bisa melewati semua part tanpa
   778→     menyentuh src (state.file uji/korup tidak membuat createObjectURL meledak) */
   779→  let _srcUrl = null;
   780→  const getSrcUrl = () => {
   781→    if (_srcUrl == null) _srcUrl = state.file.virtual ? state.file.kurl : URL.createObjectURL(state.file);
   782→    return _srcUrl;
   783→  };
   784→  const results = new Array(segs.length).fill(null);
   785→  state.exportFatal = null;   /* v2.10: mulai ekspor baru — bersihkan sisa gagal lama */
   786→  try {
   787→    if (ui.title) ui.title(segs);
   788→    if (ui.prog) ui.prog(0);
   789→    if (ui.sub) ui.sub(`encoder ${enc.kind === 'hw' ? `GPU ${enc.fps}fps` : `CPU ${enc.fps}fps`} · ${par} render paralel${RVFC_OK && !state.isAudio ? ' · turbo stream' : ''}${spd !== 1 ? ` · kecepatan ${spd}×` : ''}…`);
   790→
   791→    /* jalankan worker dengan batas `par` sekaligus (antrean)
   792→       v2.10: kegagalan part TIDAK lagi men-set state.abort (flag batal
   793→       milik pengguna) — cukup state.exportFatal supaya worker lain berhenti.
   794→       Penelepon (ekspor tunggal / batch) menerima Error aslinya dan bisa
   795→       menampilkan penyebabnya; batch menandai video GAGAL lalu LANJUT. */
   796→    let cursor = 0;
   797→    const launch = async () => {
   798→      while (cursor < slots.length && !xAborted()) {
   799→        const idx = cursor++;
   800→        const slot = slots[idx];
   801→        /* v3.0: LANJUTKAN EKSPOR — lewati part yang filenya SUDAH ADA di
   802→           folder output (hasil ekspor sebelumnya yang tertidur/gagal di
   803→           tengah). File part utuh = tanda part itu selesai dirender. */
   804→        if (state.resumeExport && !state.isAudio) {
   805→          try {
   806→            const nm0 = partFileName(state.title, state.partPrefix, slot.seg.i + 1);
   807→            const st0 = await window.kinostra.statJoin(dir, nm0);
   808→            if (st0 && st0.ok && st0.size > 51200) {
   809→              results[idx] = { name: nm0, path: st0.path, size: st0.size, skipped: true };
   810→              slot.prog = 1; slot.fr = slot.total; slot.saveProg = 1;
   811→              uiTick(true);
   812→              continue;
   813→            }
   814→          } catch (e) { }   /* statJoin tak tersedia / gagal → render normal */
   815→        }
   816→        /* v3.0: KOMPRES PINTAR — bitrate dihitung dari TARGET UKURAN per part
   817→           (bitrate = byte target ÷ durasi output − bitrate audio) */
   818→        let cfgPart = vCfg, swPart = swCfg;
   819→        if (state.targetMB > 0) {
   820→          const durP = Math.max(0.5, (slot.seg.end - slot.seg.start) / spd);
   821→          const brP = Math.round(clamp((state.targetMB * 8388608 * 0.93) / durP - 160000, 150000, 20e6));
   822→          cfgPart = { ...vCfg, bitrate: brP };
   823→          swPart = swCfg ? { ...swCfg, bitrate: brP } : null;
   824→        }
   825→        slot.working = true; uiTick(true);
   826→        try {
   827→          results[idx] = await renderPartWorker(slot.seg, getSrcUrl(), W, H, fps, br, cfgPart, aCodec, slot, () => uiTick(false));
   828→        } catch (e) {
   829→          /* v2.5: RETRY OTOMATIS 1× dengan encoder SOFTWARE — penyembuh
   830→             utama "ekspor macet di bagian akhir" (encoder GPU berhenti).
   831→             Jangan ulangi bila user membatalkan sendiri / part lain gagal.
   832→             v2.15: kegagalan jalur hardware dicatat — 2× jatuh → sisa batch
   833→             otomatis memakai software (anti-pelan, tanpa dobel kerja). */
   834→          if (hw && !xAborted()) noteHwFailure(e && e.message || String(e));
   835→          if (!xAborted() && swCfg) {
   836→            console.warn(`part ${idx + 1} gagal (${e && e.message || e}) — ulangi 1× dengan encoder software`);
   837→            slot.fr = 0; slot.stage = null; slot.saveProg = 0; slot.retried = true; slot.working = true; uiTick(true);
   838→            try {
   839→              results[idx] = await renderPartWorker(slot.seg, getSrcUrl(), W, H, fps, br, swPart || swCfg, aCodec, slot, () => uiTick(false));
   840→              console.info(`part ${idx + 1} sukses pada percobaan ke-2 (encoder software)`);
   841→            } catch (e2) {
   842→              const det2 = e2 && (e2.name + ' | ' + e2.message) || String(e2);
   843→              console.error(`part ${idx + 1} GAGAL juga di percobaan ke-2: ${det2}`);
   844→              if (!xAborted()) state.exportFatal = e2;
   845→              slot.working = false; uiTick(true);
   846→              return;
   847→            }
   848→          } else {
   849→            const det = e && (e.name + ' | ' + e.message + ' | ' + String(e.stack || '').split('\n')[1] || '') || String(e);
   850→            console.error(`part ${idx + 1} GAGAL: ${det}`);
   851→            if (!xAborted()) state.exportFatal = e;
   852→            slot.working = false; uiTick(true);
   853→            return;
   854→          }
   855→        }
   856→        slot.working = false; uiTick(true);
   857→      }
   858→    };
   859→    const runners = [];
   860→    for (let k = 0; k < par; k++) runners.push(launch());
   861→    /* pantau progres selagi worker jalan
   862→       v2.5: + WATCHDOG GLOBAL — bila TIDAK ADA kemajuan apa pun dari semua
   863→       worker selama 45 dtk, hentikan dengan pesan jelas (bukan macam) */
   864→    await new Promise(res => {
   865→      let lastSig = '', lastSigAt = Date.now();
   866→      const iv = setInterval(() => {
   867→        uiTick(false);
   868→        if (xAborted() || results.every(r => r)) { clearInterval(iv); res(); return; }
   869→        const sig = slots.map(s => `${s.fr}:${s.working ? 1 : 0}:${s.stage || ''}:${Math.round((s.saveProg || 0) * 100)}`).join('|');
   870→        const now = Date.now();
   871→        if (sig !== lastSig) { lastSig = sig; lastSigAt = now; }
   872→        else if (now - lastSigAt > 45000) {
   873→          clearInterval(iv);
   874→          /* v2.10: watchdog = kegagalan internal, BUKAN batal pengguna */
   875→          if (!xAborted()) {
   876→            state.exportFatal = new Error('Render berhenti merespons 45 detik — dihentikan otomatis. Coba set PARALEL 1× atau kualitas lebih rendah lalu ulangi.');
   877→          }
   878→          res();
   879→        }
   880→      }, 150);
   881→    });
   882→    await Promise.all(runners).catch(() => { });
   883→    if (state.exportFatal) { const err = state.exportFatal; throw err; }
   884→    if (state.abort && !results.every(r => r)) {
   885→      const made = results.filter(Boolean);
   886→      throw new Error(made.length ? `Dibatalkan — ${made.length} part tersimpan` : 'Dibatalkan');
   887→    }
   888→    if (results.some(r => !r)) throw new Error('Sebagian part gagal dirender');
   889→    if (ui.sub) {
   890→      const el = (performance.now() - t0) / 1000;
   891→      ui.sub(`selesai dalam ${el < 90 ? el.toFixed(0) + ' detik' : (el / 60).toFixed(1) + ' menit'} · ${par}× paralel${hw ? ' · GPU' : ''}`);
   892→    }
   893→    if (ui.prog) ui.prog(1);
   894→    return results;
   895→  } finally {
   896→    state.exportFatal = null;   /* v2.10: jangan bocor ke ekspor/video berikutnya */
   897→    if (_srcUrl != null && !state.file.virtual) try { URL.revokeObjectURL(_srcUrl); } catch (e) { }
   898→  }
   899→}

  1003→  await ensureFontsReady();

  1005→  const fps = state.fps, sc = parseFloat(state.scale), q = QUAL[state.quality];
  1006→  const spd = clamp(state.speed || 1, 0.25, 4);
  1007→  const dims = outDims(sc), { W, H } = dims;
  1008→  const br = Math.round(q.br * (sc === 1 ? 1 : sc === 0.75 ? 0.62 : 0.38));
  1009→  const vCfg = await pickVCodecCfg(W, H, br, fps);
  1010→  if (!vCfg) throw new Error('Tidak ada codec video yang didukung engine');
  1011→  const swCfg = await pickSWCfg(W, H, br, fps);
  1012→  const aCodec = await pickACodec();
  1013→  if (!aCodec) throw new Error('Encoder audio tidak tersedia');
  1014→  const nm = `GABUNGAN_${paths.length}_VIDEO_${slug(state.batch.find(b => b.path === paths[0])?.name || 'FILM').slice(0, 24)}.mp4`;
  1015→  /* mulai file gabungan (streaming disk — RAM datar) */
  1016→  const w = await window.kinostra.beginWrite(dir, nm);
  1017→  const mergeCtx = { muxer: null, off: 0, outName: nm, outPath: w.path, wsId: w.id, chain: Promise.resolve(), sent: 0, wsErr: null };
  1018→  mergeCtx.muxer = new Mp4Muxer.Muxer({
  1019→    target: new Mp4Muxer.StreamTarget({
  1020→      chunked: true, chunkSize: 8 * 1024 * 1024,
  1021→      onData: (data, pos) => {
  1022→        if (mergeCtx.wsErr) return;
  1023→        mergeCtx.sent += data.byteLength;
  1024→        mergeCtx.chain = mergeCtx.chain.then(() => window.kinostra.writeChunkAt(mergeCtx.wsId, data, pos)).catch(e => { mergeCtx.wsErr = mergeCtx.wsErr || e; });
  1025→      }
  1026→    }),
  1027→    fastStart: false, firstTimestampBehavior: 'offset',
  1028→    video: { codec: 'avc', width: W, height: H, frameRate: fps },
  1029→    audio: { codec: aCodec === 'mp4a.40.2' ? 'aac' : 'opus', numberOfChannels: 2, sampleRate: 48000 }
  1030→  });
  1031→  state._noPart = true;          /* label PART tidak masuk file gabungan */
  1032→  const prevFatal = state.exportFatal;
  1033→  state.exportFatal = null;
  1034→  const t0 = performance.now();
  1035→  let doneParts = 0;
  1036→  try {
  1037→    if (ui.prog) ui.prog(0);
  1038→    if (ui.sub) ui.sub(`Menyiapkan encoder ${_lastEnc.kind === 'hw' ? 'GPU' : 'CPU'} · ${paths.length} video…`);
  1039→    for (let i = 0; i < paths.length; i++) {
  1040→      if (state.abort) throw new Error('Dibatalkan');
  1041→      if (state.exportFatal) throw state.exportFatal;
  1042→      if (mergeCtx.wsErr) throw mergeCtx.wsErr;
  1043→      const nmv = paths[i].split(/[\\/]/).pop();
  1044→      if (ui.sub) ui.sub(`[${i + 1}/${paths.length}] Memuat ${nmv}…`);
  1045→      /* lepaskan buffer film sebelumnya (hemat RAM, sama dgn batch) */
  1046→      state.audioBuffer = null; state.peaks = null; state.audioRms = 0;
  1047→      await loadFromPath(paths[i], true);
  1048→      if (!state.file || state.file.path !== paths[i]) throw new Error(`video ${i + 1} (${nmv}) tidak bisa dimuat`);
  1049→      if (!state.duration) throw new Error(`durasi ${nmv} tidak terbaca — format tak didukung`);
  1050→      if (ui.sub) ui.sub(`[${i + 1}/${paths.length}] ${nmv} · ${fmtT(state.duration)} · memeriksa subtitel…`);
  1051→      await ensureAutoSubs({ sub: s => { if (ui.sub) ui.sub(`[${i + 1}/${paths.length}] ${s}`); }, prog: () => { } });
  1052→      if (state.music.buffer) await composeMusic(false, true, true);
  1053→      const srcUrl = state.file.virtual ? state.file.kurl : URL.createObjectURL(state.file);
  1054→      try {
  1055→        const segs = segments();
  1056→        for (const seg of segs) {
  1057→          if (state.abort) throw new Error('Dibatalkan');
  1058→          if (state.exportFatal) throw state.exportFatal;
  1059→          const slot = { seg, dir, prog: 0, fr: 0, total: Math.max(1, Math.round((seg.end - seg.start) / spd * fps)), working: true, stage: null, saveProg: 0 };
  1060→          try {
  1061→            await renderPartWorker(seg, srcUrl, W, H, fps, br, vCfg, aCodec, slot, () => {
  1062→              doneParts += 0.0001;
  1063→              if (ui.prog) ui.prog(clamp((mergeCtx.off + (seg.end - seg.start)) / Math.max(0.1, expLen() || 1) * 0, 0, 1)); /* prog pakai fr di bawah */
  1064→            }, mergeCtx);
  1065→          } catch (e) {
  1066→            if (state.acc === 'gpu' && swCfg && !xAborted()) {
  1067→              console.warn(`merge part gagal (${e && e.message || e}) — ulangi dengan encoder software`);
  1068→              await renderPartWorker(seg, srcUrl, W, H, fps, br, swCfg, aCodec, slot, null, mergeCtx);
  1069→            } else throw e;
  1070→          }
  1071→          mergeCtx.off += (seg.end - seg.start) / spd;
  1072→          if (ui.prog) ui.prog(clamp(mergeCtx.off / Math.max(0.1, (state.duration || 1) * paths.length), 0, 1));
  1073→          if (ui.sub) ui.sub(`[${i + 1}/${paths.length}] ${nmv} · part ${slot.seg.i + 1}/${segs.length} selesai · total ${fmtT(mergeCtx.off)}`);
  1074→        }
  1075→      } finally {
  1076→        if (!state.file.virtual) try { URL.revokeObjectURL(srcUrl); } catch (e) { }
  1077→      }
  1078→    }
  1079→    if (state.abort) throw new Error('Dibatalkan');
  1080→    if (ui.sub) ui.sub('Menuntaskan header MP4…');
  1081→    await withTimeout(mergeCtx.chain, 180000, 'menulis file gabungan macet (cek ruang disk)');
  1082→    if (mergeCtx.wsErr) throw mergeCtx.wsErr;
  1083→    await withTimeout(Promise.resolve().then(() => mergeCtx.muxer.finalize()), 60000, 'penulisan header MP4 gabungan macet');
  1084→    await window.kinostra.endWrite(mergeCtx.wsId); mergeCtx.wsId = null;
  1085→    const stt = await window.kinostra.stat(mergeCtx.outPath);
  1086→    const el = (performance.now() - t0) / 1000;
  1087→    if (ui.sub) ui.sub(`selesai dalam ${el < 90 ? el.toFixed(0) + ' detik' : (el / 60).toFixed(1) + ' menit'}`);
  1088→    return [{ name: nm, path: mergeCtx.outPath, size: stt && stt.ok ? stt.size : mergeCtx.sent }];
  1089→  } catch (e) {
  1090→    /* gagal → buang file setengah jadi */
  1091→    try { if (mergeCtx.wsId) { const id = mergeCtx.wsId; mergeCtx.wsId = null; await window.kinostra.abortWrite(id); } } catch (e2) { }
  1092→    throw e;
  1093→  } finally {
  1094→    state._noPart = false;
  1095→    state.exportFatal = prevFatal;
  1096→  }
  1097→}
